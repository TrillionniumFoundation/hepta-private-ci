def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def identity(value):
    return {name: getattr(value, name) for name in (
        "st_dev", "st_ino", "st_mode", "st_nlink", "st_uid", "st_gid",
        "st_size", "st_mtime_ns", "st_ctime_ns"
    )}

def version(value):
    return ":".join(str(getattr(value, name)) for name in (
        "st_dev", "st_ino", "st_size", "st_mtime_ns", "st_ctime_ns"
    ))

def regular(value):
    if not stat.S_ISREG(value.st_mode):
        raise ValueError("Only a regular nonsymlink file is permitted")
    return value

def read_original():
    before = regular(os.lstat(SOURCE))
    fd = os.open(SOURCE, os.O_RDONLY | NOFOLLOW | CLOEXEC)
    try:
        opened = regular(os.fstat(fd))
        if identity(before) != identity(opened):
            raise ValueError("Original path and descriptor differ")
        if before.st_size != EXPECTED_SIZE or version(before) != EXPECTED_VERSION:
            raise ValueError("Original size or version differs from the observed pin")
        blocks = []
        remaining = EXPECTED_SIZE + 1
        while remaining:
            block = os.read(fd, min(1048576, remaining))
            if not block:
                break
            blocks.append(block)
            remaining -= len(block)
        raw = b"".join(blocks)
        after_fd = regular(os.fstat(fd))
        after_path = regular(os.lstat(SOURCE))
        if any(identity(s) != identity(before) for s in (after_fd, after_path)):
            raise ValueError("Original changed while being read")
        if len(raw) != EXPECTED_SIZE or sha(raw) != EXPECTED_SHA256:
            raise ValueError("Original bytes differ from the expected content receipt")
        return fd, before, raw
    except BaseException:
        os.close(fd)
        raise

def require_original_unchanged(fd, before):
    current_fd = regular(os.fstat(fd))
    current_path = regular(os.lstat(SOURCE))
    if identity(current_fd) != identity(before) or identity(current_path) != identity(before):
        raise ValueError("Original path or descriptor changed during archiving")

def read_regular(path, limit):
    before = regular(os.lstat(path))
    if before.st_size > limit:
        raise ValueError("Existing archive member exceeds its bound")
    fd = os.open(path, os.O_RDONLY | NOFOLLOW | CLOEXEC)
    with os.fdopen(fd, "rb") as stream:
        if identity(regular(os.fstat(stream.fileno()))) != identity(before):
            raise ValueError("Existing archive path and descriptor differ")
        raw = stream.read(limit + 1)
        if len(raw) > limit or identity(os.fstat(stream.fileno())) != identity(before):
            raise ValueError("Existing archive member changed or exceeded its bound")
    if identity(regular(os.lstat(path))) != identity(before):
        raise ValueError("Existing archive path changed")
    return raw, before

def create_only(path, raw):

    with os.fdopen(fd, "wb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
