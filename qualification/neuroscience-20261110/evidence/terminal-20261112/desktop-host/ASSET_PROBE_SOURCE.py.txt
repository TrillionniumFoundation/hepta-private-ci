def small(path, limit=65536):
    try:
        with open(path, "rb") as stream:
            value = stream.read(limit + 1)
        return value[:limit].decode("utf-8", "replace"), len(value) > limit
    except OSError:
        return None, False

def redacted_remote(raw):
    host, path = None, ""
    if "://" in raw:
        try:
            value = urllib.parse.urlsplit(raw)
            host, path = value.hostname, value.path
        except ValueError:
            pass
    else:
        match = re.fullmatch(r"(?:[^@\s]+@)?([^:/\s]+):(.+)", raw)
        if match:
            host, path = match.groups()
    match = path.rstrip("/").removesuffix(".git").lower().endswith("/trillionniumfoundation/hepta-private-ci") or path.rstrip("/").removesuffix(".git").lower() == "trillionniumfoundation/hepta-private-ci"
    return {"host":host, "repository":"TrillionniumFoundation/hepta-private-ci" if match else "NON_TARGET_OR_UNRECOGNIZED", "expected_repository_name":match, "raw_url_printed":False}
