function scanTupleSources(sources,rows){
 function expr(s,v){s=s.trim();let out="";while(s.length){s=s.trimStart();const q=s[0];if(q==="'"||q==='"'){let i=1,x="";while(i<s.length){let c=s[i++];if(c===q)break;if(c==="\\"){c=s[i++];x+=({n:"\n",r:"\r",t:"\t","\\":"\\","'":"'",'"':'"'}[c]??c);}else x+=c;}out+=x;s=s.slice(i);}else{const m=s.match(/^[A-Z_][A-Z_0-9]*/);if(!m||!Object.hasOwn(v,m[0]))return null;out+=v[m[0]];s=s.slice(m[0].length);}s=s.trimStart();if(!s.length||s[0]==="#")return out;if(s[0]!=="+")return null;s=s.slice(1);}return out;}
 const checks=[];for(const f of sources){const c=f.content,v={};for(let pass=0;pass<4;pass++)for(const m of c.matchAll(/^([A-Z_][A-Z_0-9]*)\s*=\s*(.+)$/gm)){const t=expr(m[2],v);if(t!==null)v[m[1]]=t;}
 for(const row of rows){const re=new RegExp('(["\\\'])'+row.id+'\\1\\s*:\\s*([\\(\\[])',"g");let m;while((m=re.exec(c))){const start=m.index+m[0].length-1,close=c[start]==="("?")":"]";let q=null,esc=false,depth=0,end=-1;for(let i=start;i<c.length;i++){const z=c[i];if(q){if(esc)esc=false;else if(z==="\\")esc=true;else if(z===q)q=null;continue;}if(z==="'"||z==='"'){q=z;continue;}if(z===c[start])depth++;if(z===close&&--depth===0){end=i+1;break;}}if(end<0)throw Error("unclosed tuple");const body=c.slice(start+1,end-1),parts=[];let q2=null,e=false,last=0;for(let i=0;i<body.length;i++){const z=body[i];if(q2){if(e)e=false;else if(z==="\\")e=true;else if(z===q2)q2=null;}else if(z==="'"||z==='"')q2=z;else if(z===","){parts.push(body.slice(last,i));last=i+1;}}parts.push(body.slice(last));const actual=parts.filter(x=>x.trim()).map(x=>expr(x,v));checks.push({path:f.path,blob:f.sha,id:row.id,line:c.slice(0,start).split("\n").length,actual,expected:row.product_callers,equal:JSON.stringify(actual)===JSON.stringify(row.product_callers),unresolved:actual.some(x=>x===null)});re.lastIndex=end;}}}return checks;
}
