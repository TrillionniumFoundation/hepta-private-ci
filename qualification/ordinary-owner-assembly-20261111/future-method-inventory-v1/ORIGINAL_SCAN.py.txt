entries={}
for item in subprocess.check_output(["git","ls-tree","-r","-z",head,"--","codex-rs"],cwd=root).split(b"\0"):

    meta,name=item.split(b"\t",1);mode,kind,blob=meta.decode().split();name=name.decode()

    files.append({"path":name,"git_blob":actual,"bytes":len(data),"sha256":hashlib.sha256(data).hexdigest()})
    for match in matches:
        line=content.count("\n",0,match.start())+1
        start=content.rfind("\n",0,match.start())+1;end=content.find("\n",match.end())
        if end<0:end=len(content)
        hits[match.group(1)].append({"path":name,"line":line,"text":content[start:end]})
