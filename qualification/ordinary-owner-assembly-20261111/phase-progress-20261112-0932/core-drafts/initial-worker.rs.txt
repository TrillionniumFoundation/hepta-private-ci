//! Late original enrollment and one initial connection share the exact
//! lifetime of the original Root listeners, not a detached replacement service.

use std::net::Shutdown;
use std::os::unix::net::UnixListener;
use std::os::unix::net::UnixStream;
use std::sync::atomic::AtomicBool;

use super::*;

pub(in super::super) struct BoundInitial {
    runtime: InitialRuntime,
    listener: UnixListener,
    device: u64,
    inode: u64,
}

impl BoundInitial {
    pub(in super::super) fn bind(
        runtime: InitialRuntime,
        config: &HostAuthorityServiceConfig,
    ) -> io::Result<Self> {
        let mut paths: Vec<_> = validate_runtime_lanes(config)?
            .into_iter()
            .map(|(_, path)| path)
            .collect();
        paths.push(&runtime.socket);
        validate_lane_paths(&paths, config.peer_gid)?;
        let listener = UnixListener::bind(&runtime.socket)?;
        let metadata = std::fs::symlink_metadata(&runtime.socket)?;
        let bound = Self {
            runtime,
            listener,
            device: metadata.dev(),
            inode: metadata.ino(),
        };
        if metadata.uid() != 0 || metadata.gid() != config.peer_gid {
            return Err(io::Error::other("initial listener custody differs"));
        }
        std::fs::set_permissions(
            &bound.runtime.socket,
            std::fs::Permissions::from_mode(0o660),
        )?;
        bound.listener.set_nonblocking(true)?;
        Ok(bound)
    }

    pub(in super::super) fn path(&self) -> &Path {
        &self.runtime.socket
    }
}

impl Drop for BoundInitial {
    fn drop(&mut self) {
        self.runtime.phase.store(CLOSED, Ordering::Release);
        if std::fs::symlink_metadata(&self.runtime.socket)
            .is_ok_and(|entry| entry.dev() == self.device && entry.ino() == self.inode)
        {
            let _ = std::fs::remove_file(&self.runtime.socket);
        }
    }
}

pub(in super::super) fn serve_optional(
    initial: Option<&BoundInitial>,
    service: &HostAuthorityService,
    config: &HostAuthorityServiceConfig,
    registration: &HostAuthorityRegistration,
    owner: &OwnedFd,
    original: impl FnOnce() -> io::Result<()>,
) -> io::Result<()> {
    let Some(initial) = initial else {
        return original();
    };
    let stopping = AtomicBool::new(false);
    let active = Mutex::new(None::<UnixStream>);
    let end = Instant::now() + Duration::from_millis(config.lifetime_ms);
    initial
        .runtime
        .phase
        .compare_exchange(PREPARED, ACTIVE, Ordering::AcqRel, Ordering::Acquire)
        .map_err(|_| io::Error::other("original initial runtime already consumed"))?;
    std::thread::scope(|scope| {
        let worker = scope.spawn(|| {
            let receiver = initial
                .runtime
                .receiver
                .try_lock()
                .map_err(|_| io::Error::other("initial enrollment receiver busy"))?;
            let submission = loop {
                if stopping.load(Ordering::Acquire) {
                    return Ok(());
                }
                owner_alive(owner, end)?;
                match receiver.try_recv() {
                    Ok(submission) => break submission,
                    Err(mpsc::TryRecvError::Empty) => std::thread::sleep(Duration::from_millis(5)),
                    Err(mpsc::TryRecvError::Disconnected) => return Ok(()),
                }
            };
            let slot = service.enroll_generation_initial(
                config,
                registration,
                owner,
                &initial.runtime.deployment,
                submission.inputs,
            );
            let slot = match slot {
                Ok(slot) => {
                    if stopping.load(Ordering::Acquire)
                        || submission
                            .acknowledgement
                            .try_send(Ok(slot.profile.binding.clone()))
                            .is_err()
                    {
                        return Err(io::Error::other(
                            "original initial enrollment acknowledgement lost",
                        ));
                    }
                    slot
                }
                Err(error) => {
                    let _ = submission.acknowledgement.try_send(Err(error.to_string()));
                    return Err(error);
                }
            };
            let stream = loop {
                if stopping.load(Ordering::Acquire) {
                    return Ok(());
                }
                owner_alive(owner, end)?;
                match initial.listener.accept() {
                    Ok((stream, _)) => break stream,
                    Err(error) if error.kind() == io::ErrorKind::WouldBlock => {
                        std::thread::sleep(Duration::from_millis(5));
                    }
                    Err(error) => return Err(error),
                }
            };
            verify_pinned_process(registration, config.peer_uid, config.peer_gid, owner, end)?;
            {
                let mut held = active
                    .lock()
                    .map_err(|_| io::Error::other("initial lane poisoned"))?;
                if stopping.load(Ordering::Acquire) {
                    return Ok(());
                }
                *held = Some(stream.try_clone()?);
            }
            let policy = HostAuthorityPeerPolicy {
                peer: HostIssuerPeer {
                    uid: config.peer_uid,
                    pid: registration.peer_pid,
                },
                timeout: Duration::from_millis(config.timeout_ms),
                lifetime: end
                    .checked_duration_since(Instant::now())
                    .ok_or_else(|| io::Error::other("initial Root session expired"))?,
                request_bound: 16 * 1024,
                response_bound: 16 * 1024,
            };
            let result = service.serve_generation_initial(
                &slot,
                config,
                registration,
                stream,
                policy,
                owner.try_clone()?,
            );
            if stopping.load(Ordering::Acquire) {
                Ok(())
            } else {
                result
            }
        });
        let result = original();
        stopping.store(true, Ordering::Release);
        initial.runtime.phase.store(CLOSED, Ordering::Release);
        if let Some(stream) = active
            .lock()
            .map_err(|_| io::Error::other("initial lane poisoned"))?
            .as_ref()
        {
            let _ = stream.shutdown(Shutdown::Both);
        }
        let served = worker
            .join()
            .map_err(|_| io::Error::other("initial lane panicked"))?;
        result.and(served)
    })
}

impl BoundInitial {
    pub(in super::super) fn activate_same_actor_phase(&self) -> io::Result<()> {
        self.runtime.phase.compare_exchange(PREPARED, ACTIVE, Ordering::AcqRel, Ordering::Acquire)
            .map_err(|_| io::Error::other("original phase Initial runtime already consumed"))?;
        Ok(())
    }

    /// Transfer the held Initial slot only after a pre-authorized retirement
    /// and clean EOF. The phase worker still requires the separate completed
    /// join, validates the real physical cut, durably claims, and drops this
    /// owner before opening successor CURRENT. EOF alone grants no transition.
    pub(in super::super) fn serve_same_actor_phase_initial(
        &self,
        service: &HostAuthorityService,
        config: &HostAuthorityServiceConfig,
        registration: &HostAuthorityRegistration,
        owner: &OwnedFd,
        gate: &Arc<crate::same_actor_q24_phase_gate::SameActorPhaseGate>,
        retirement: &crate::authority_host_same_actor_q24_phase_join::OriginalInitialRetirement,
    ) -> io::Result<crate::authority_host_service::generation_initial::RegisteredGenerationInitial> {
        let mut closing = crate::same_actor_q24_phase_gate::CloseSameActorPhaseOnDrop::new(gate);
        let end = gate.deadline().instant();
        let receiver = self.runtime.receiver.try_lock()
            .map_err(|_| io::Error::other("original phase Initial receiver busy"))?;
        let submission = loop {
            let epoch = gate.original_wait_epoch()?;
            owner_alive(owner, end)?;
            match receiver.try_recv() {
                Ok(submission) => break submission,
                Err(mpsc::TryRecvError::Empty) => gate.wait_original_change(epoch)?,
                Err(mpsc::TryRecvError::Disconnected) => {
                    return Err(io::Error::other("original phase Initial capability was dropped"));
                }
            }
        };
        if submission.inputs.mode != crate::RootGenerationInitialModeV1::Reserve {
            return Err(io::Error::other("same-actor phase requires the original fresh A"));
        }
        let slot = service.enroll_generation_initial(
            config, registration, owner, &self.runtime.deployment, submission.inputs,
        );
        let slot = match slot {
            Ok(slot) => {
                gate.require_open()?;
                submission.acknowledgement.try_send(Ok(slot.profile.binding.clone()))
                    .map_err(|_| io::Error::other("original phase Initial acknowledgement lost"))?;
                gate.notify_original_waiters();
                slot
            }
            Err(error) => {
                let _ = submission.acknowledgement.try_send(Err(error.to_string()));
                gate.notify_original_waiters();
                return Err(error);
            }
        };
        let stream = loop {
            let epoch = gate.original_wait_epoch()?;
            owner_alive(owner, end)?;
            match self.listener.accept() {
                Ok((stream, _)) => break stream,
                Err(error) if error.kind() == io::ErrorKind::WouldBlock => {
                    gate.wait_original_change(epoch)?;
                }
                Err(error) => return Err(error),
            }
        };
        verify_pinned_process(registration, config.peer_uid, config.peer_gid, owner, end)?;
        let policy = HostAuthorityPeerPolicy {
            peer: HostIssuerPeer { uid: config.peer_uid, pid: registration.peer_pid },
            timeout: Duration::from_millis(config.timeout_ms),
            lifetime: Duration::from_millis(config.lifetime_ms),
            request_bound: 16 * 1024,
            response_bound: 16 * 1024,
        };
        service.serve_generation_initial_until(
            &slot, config, registration, stream, policy, owner.try_clone()?, gate,
        )?;
        retirement.require_expected_eof(gate)?;
        closing.disarm();
        Ok(slot)
    }
}
