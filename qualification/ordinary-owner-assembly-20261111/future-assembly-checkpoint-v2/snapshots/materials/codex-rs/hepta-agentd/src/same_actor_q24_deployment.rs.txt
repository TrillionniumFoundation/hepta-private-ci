pub(crate) use owner::OriginalDeploymentCoordinatorOwner;

/// Only a selected pathname and exact immutable digest, not authority.
pub(crate) struct OriginalDeploymentDescriptor(files::PinnedFileData);

impl OriginalDeploymentDescriptor {
    pub(crate) fn from_original_arguments(path: PathBuf, digest: Digest32) -> Result<Self, AgentdError> {
        if !path.is_absolute() {
            return Err(deployment_refused("original deployment descriptor must be absolute"));
        }
        Ok(Self(files::PinnedFileData { path, sha256: digest.into_array() }))
    }
}

pub(crate) async fn open_original_deployment(
    context: Arc<OriginalSameActorRootContext>,
    descriptor: OriginalDeploymentDescriptor,
) -> Result<(OriginalSameActorBusinessInputsV1, OriginalDeploymentCoordinatorOwner), AgentdError> {
    let factory::OriginalLoadedDeployment { business, coordinator } =
        factory::load_original_deployment(Arc::clone(&context), descriptor.0).await?;
    let owner = OriginalDeploymentCoordinatorOwner::start_original(context, coordinator)?;
    Ok((business, owner))
}

fn deployment_refused(message: &str) -> AgentdError {
    AgentdError::GenerationFenced(format!("original same-actor deployment: {message}"))
}

fn deployment_error(error: impl std::fmt::Display) -> AgentdError {
    AgentdError::Invalid(format!("original same-actor deployment: {error}"))
}
