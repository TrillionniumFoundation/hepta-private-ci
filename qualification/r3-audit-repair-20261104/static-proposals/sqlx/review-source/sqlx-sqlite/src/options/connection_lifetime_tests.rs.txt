const BOUND: Duration = Duration::from_secs(5);
const CHILD: &str = "options::connection_lifetime::tests::owned_scenario";
static SERIAL: AtomicUsize = AtomicUsize::new(0);

#[derive(Debug)]
enum LockState {
    Held,
    Released,
}

enum GateState { Absent, Blocked, Released }

struct OwnedScenario {
    child: Child,
    input: Option<ChildStdin>,
    lines: Option<mpsc::Receiver<String>>,
    reader: Option<JoinHandle<()>>,
    directory: PathBuf,
    reaped: bool,
}

impl OwnedScenario {
    fn start(scenario: &str) -> Self {
        let directory = std::env::temp_dir().join(format!(
            "sqlx-native-custody-{}-{}",
            std::process::id(),
            SERIAL.fetch_add(1, Ordering::Relaxed)
        ));
        fs::create_dir(&directory).unwrap();
        let mut child = Command::new(std::env::current_exe().unwrap())
            .args(["--exact", CHILD, "--ignored", "--nocapture", "--test-threads=1"])
            .env("SQLX_LIFETIME_SCENARIO", scenario)
            .env("SQLX_LIFETIME_DIRECTORY", &directory)
            .stdin(Stdio::piped())
            .stdout(Stdio::piped())
            .stderr(Stdio::inherit())
            .spawn()
            .unwrap();
        let input = child.stdin.take();
        let output = child.stdout.take().unwrap();
        let (sender, lines) = mpsc::sync_channel(64);
        let mut owned = Self {
            child, input, lines: Some(lines), reader: None, directory, reaped: false,
        };
        owned.reader = Some(thread::spawn(move || {
            for line in BufReader::new(output).lines() {
                match line {
                    Ok(line) => { if sender.send(line).is_err() { break; } }
                    Err(_) => break,
                }
            }
        }));
        owned
    }

    fn marker(&self) -> String {
        let deadline = Instant::now() + BOUND;
        loop {
            let line = self.lines.as_ref().unwrap()
                .recv_timeout(deadline.saturating_duration_since(Instant::now()))
                .expect("owned scenario did not reach its physical checkpoint");
            if let Some(stage) = line.strip_prefix("LIFETIME: ") {
                eprintln!("owned lifetime checkpoint={stage}");
                return stage.to_owned();
            }
        }
    }

    fn send(&mut self, command: &[u8]) {
        self.input.as_mut().unwrap().write_all(command).unwrap();
        self.input.as_mut().unwrap().flush().unwrap();
    }

    fn checkpoint(&mut self, expected: &str, lock: LockState, gate: GateState) {
        assert_eq!(self.marker(), expected);
        let actual = self.probe_lock();
        eprintln!("physical custody expected={lock:?} result={actual:?}");
        self.send(b"inspect\n");
        let expected_gate = match gate {
            GateState::Absent => "gate_status=-1",
            GateState::Blocked => "gate_status=0",
            GateState::Released => "gate_status=1",
        };
        // Check after the physical probe so a timed-out callback cannot count
        // as the intended early-release counterfactual.
        assert_eq!(self.marker(), expected_gate, "native gate is not in the required state");
        match (lock, actual) {
            (LockState::Held, Err(TryLockError::WouldBlock))
            | (LockState::Released, Ok(())) => {}
            (_, actual) => panic!("unexpected physical lock result: {actual:?}"),
        }
        self.send(b"go\n");
    }

    fn probe_lock(&self) -> Result<(), TryLockError> {
        let contender = OpenOptions::new().read(true).write(true)
            .open(self.directory.join("custody.lock")).unwrap();
        contender.try_lock()
    }

    fn finish(&mut self) {
        let deadline = Instant::now() + BOUND;
        loop {
            if let Some(status) = self.child.try_wait().unwrap() {
                self.reaped = true;
                assert!(status.success(), "owned scenario failed: {status}");
                assert!(self.probe_lock().is_ok(), "custody survived owned process exit");
                return;
            }
            assert!(Instant::now() < deadline, "owned scenario did not exit");
            thread::sleep(Duration::from_millis(10));
        }
    }
}

impl Drop for OwnedScenario {
    fn drop(&mut self) {
        drop(self.input.take());
        drop(self.lines.take());
        if !self.reaped {
            let _ = self.child.kill();
            let _ = self.child.wait();
        }
        if let Some(reader) = self.reader.take() {
            let _ = reader.join();
        }
        let _ = fs::remove_dir_all(&self.directory);
    }
}

#[test]
fn cancellation_during_native_open_keeps_resource_custody() {
    let mut child = OwnedScenario::start("open");
    child.checkpoint("cancelled_inside_native_open", LockState::Held, GateState::Blocked);
    child.checkpoint("native_closed", LockState::Released, GateState::Released);
    child.finish();
}

#[test]
fn cancellation_during_initial_pragmas_keeps_resource_custody() {
    let mut child = OwnedScenario::start("pragma");
    child.checkpoint("cancelled_inside_initial_pragma", LockState::Held, GateState::Blocked);
    child.checkpoint("native_closed", LockState::Released, GateState::Released);
    child.finish();
}

#[test]
fn pool_growth_and_cancelled_close_keep_borrowed_connection_custody() {
    let mut child = OwnedScenario::start("pool");
    child.checkpoint("closed_pool_with_live_detached_connection", LockState::Held, GateState::Absent);
    child.checkpoint("native_closed", LockState::Released, GateState::Absent);
    child.finish();
}

#[test]
fn initialization_error_releases_but_failed_native_close_retains_custody() {
    let mut child = OwnedScenario::start("failure");
    child.checkpoint("initialization_error_preserved_and_closed", LockState::Released, GateState::Absent);
    child.checkpoint("busy_close_failed_and_native_statement_still_runs", LockState::Held, GateState::Absent);
    child.checkpoint("manual_native_close_cannot_release_forgotten_custody", LockState::Held, GateState::Absent);
    child.finish();
}

// Explicitly invoked by the four parents using the ordinary libtest main.
// This support entry is not a fifth product control or a passed ignored test.
#[test]
#[ignore]
fn owned_scenario() {
    let scenario = std::env::var("SQLX_LIFETIME_SCENARIO").unwrap();
    let directory = PathBuf::from(std::env::var_os("SQLX_LIFETIME_DIRECTORY").unwrap());
    match scenario.as_str() {
        "open" | "pragma" => cancelled_initialization(&directory, &scenario),
        "pool" => pool_lifetime(&directory),
        "failure" => error_lifetime(&directory),
        _ => panic!("unknown owned scenario"),
    }
}

struct ResourceOwner {
    file: Option<File>,
    released: mpsc::Sender<()>,
}

impl Drop for ResourceOwner {
    fn drop(&mut self) {
        drop(self.file.take());
        let _ = self.released.send(());
    }
}

fn options(directory: &Path) -> (SqliteConnectOptions, mpsc::Receiver<()>) {
    let file = OpenOptions::new().read(true).write(true).create(true)
        .truncate(false).open(directory.join("custody.lock")).unwrap();
    file.lock().unwrap();
    let (released, receiver) = mpsc::channel();
    let token = Arc::new(ResourceOwner { file: Some(file), released });
    let options = SqliteConnectOptions::new()
        .filename(directory.join("database.sqlite"))
        .create_if_missing(true)
        .connection_lifetime_token(token);
    (options, receiver)
}

fn checkpoint(stage: &str) {
    println!("\nLIFETIME: {stage}");
    std::io::stdout().flush().unwrap();
    let mut line = String::new();
    std::io::stdin().read_line(&mut line).unwrap();
    assert_eq!(line, "inspect\n");
    let status = NATIVE_GATE.get().map(|gate| gate.outcome.load(Ordering::Acquire)).unwrap_or(-1);
    println!("LIFETIME: gate_status={status}");
    std::io::stdout().flush().unwrap();
    line.clear();
    std::io::stdin().read_line(&mut line).unwrap();
    assert_eq!(line, "go\n");
}

enum GatePoint {
    Open,
    Pragma,
}

struct NativeGate {
    point: GatePoint,
    entered: flume::Sender<()>,
    release: Mutex<mpsc::Receiver<()>>,
    first_progress: AtomicBool,
    outcome: AtomicI32,
}

static NATIVE_GATE: OnceLock<NativeGate> = OnceLock::new();

fn wait_at_native_gate() -> c_int {
    let Some(gate) = NATIVE_GATE.get() else { return 1 };
    if gate.entered.send(()).is_err() { return 1; }
    let released = match gate.release.lock() {
        Ok(receiver) => receiver.recv_timeout(BOUND).is_ok(),
        Err(_) => false,
    };
    gate.outcome.store(if released { 1 } else { 2 }, Ordering::Release);
    if released { 0 } else { 1 }
}

unsafe extern "C" fn progress(_: *mut c_void) -> c_int {
    let Some(gate) = NATIVE_GATE.get() else { return 1 };
    if gate.first_progress.swap(false, Ordering::AcqRel) {
        wait_at_native_gate()
    } else {
        0
    }
}

unsafe extern "C" fn opened(
    database: *mut sqlite3,
    _: *mut *mut c_char,
    _: *const sqlite3_api_routines,
) -> c_int {
    match NATIVE_GATE.get().map(|gate| &gate.point) {
        Some(GatePoint::Open) => wait_at_native_gate(),
        Some(GatePoint::Pragma) => {
            unsafe { sqlite3_progress_handler(database, 1, Some(progress), std::ptr::null_mut()); }
            SQLITE_OK
        }
        None => 1,
    }
}

fn cancelled_initialization(directory: &Path, scenario: &str) {
    let (entered, receiver) = flume::bounded(1);
    let (release, release_receiver) = mpsc::channel();
    let point = if scenario == "open" { GatePoint::Open } else { GatePoint::Pragma };
    assert!(NATIVE_GATE.set(NativeGate {
        point, entered, release: Mutex::new(release_receiver),
        first_progress: AtomicBool::new(true), outcome: AtomicI32::new(0),
    }).is_ok());
    assert_eq!(unsafe { sqlite3_auto_extension(Some(opened)) }, SQLITE_OK);
    let (options, closed) = options(directory);
    let mut connecting = Box::pin(options.connect());
    sqlx_core::rt::test_block_on(async {
        // Keep polling through establish so initial PRAGMAs actually execute.
        let entered = futures_util::future::select(
            connecting.as_mut(),
            Box::pin(receiver.recv_async()),
        );
        match sqlx_core::rt::timeout(BOUND, entered).await.expect("real SQLite callback did not enter") {
            futures_util::future::Either::Right((Ok(()), _)) => {}
            futures_util::future::Either::Right((Err(error), _)) => panic!("native gate disconnected: {error}"),
            futures_util::future::Either::Left((result, _)) => panic!("connect completed before the native gate: {result:?}"),
        }
    });
    drop(connecting);
    drop(options);
    checkpoint(if scenario == "open" {
        "cancelled_inside_native_open"
    } else {
        "cancelled_inside_initial_pragma"
    });
    release.send(()).unwrap();
    closed.recv_timeout(BOUND).expect("native connection did not release custody");
    assert_eq!(unsafe { sqlite3_cancel_auto_extension(Some(opened)) }, 1);
    checkpoint("native_closed");
}

fn pool_lifetime(directory: &Path) {
    let (options, closed) = options(directory);
    sqlx_core::rt::test_block_on(async {
        let pool = SqlitePoolOptions::new().min_connections(0).max_connections(5)
            .connect_lazy_with(options);
        let mut held = Vec::new();
        for _ in 0..5 {
            let mut connection = pool.acquire().await.unwrap();
            connection.execute("CREATE TEMP TABLE lifetime_marker(value);")
                .await.unwrap();
            held.push(connection);
        }
        held.pop().unwrap().close().await.unwrap();
        let mut replacement = pool.acquire().await.unwrap();
        let row = replacement.fetch_one(
            "SELECT count(*) FROM sqlite_temp_master WHERE name = 'lifetime_marker';"
        ).await.unwrap();
        assert_eq!(row.get::<i64, _>(0), 0);
        let mut closing = Box::pin(pool.close());
        assert!(matches!(closing.as_mut().poll(&mut Context::from_waker(&noop_waker())), Poll::Pending));
        drop(closing);
        for connection in held { connection.close().await.unwrap(); }
        let mut detached = replacement.detach();
        drop(pool);
        let row = detached.fetch_one("SELECT 1;").await.unwrap();
        assert_eq!(row.get::<i64, _>(0), 1);
        checkpoint("closed_pool_with_live_detached_connection");
        detached.close().await.unwrap();
    });
    closed.recv_timeout(BOUND).expect("closed physical connections retained custody");
    checkpoint("native_closed");
}

fn error_lifetime(directory: &Path) {
    let (bad_options, closed) = options(directory);
    let bad_options = bad_options.pragma("SQLX_LIFETIME_INVALID(", "1");
    let error = match futures_executor::block_on(bad_options.connect()) {
        Ok(_) => panic!("invalid initial PRAGMA unexpectedly succeeded"),
        Err(error) => error,
    };
    eprintln!("original initialization error={error:?}");
    assert_eq!(error.as_database_error().and_then(|error| error.code()).as_deref(), Some("1"));
    drop(bad_options);
    closed.recv_timeout(BOUND).expect("initialization error did not close SQLite");
    checkpoint("initialization_error_preserved_and_closed");

    let (options, _closed) = options(directory);
    let mut connection = futures_executor::block_on(options.connect()).unwrap();
    drop(options);
    let mut locked = futures_executor::block_on(connection.lock_handle()).unwrap();
    let database = locked.as_raw_handle().as_ptr();
    let mut statement = std::ptr::null_mut();
    assert_eq!(unsafe {
        sqlite3_prepare_v2(database, c"SELECT 1".as_ptr(), -1, &mut statement, std::ptr::null_mut())
    }, SQLITE_OK);
    // A genuine live statement makes native close return BUSY without destroying the handle.
    assert_eq!(unsafe { sqlite3_close(database) }, SQLITE_BUSY);
    drop(locked);
    let error = futures_executor::block_on(connection.close()).unwrap_err();
    eprintln!("original failed-close result={error:?}");
    assert!(matches!(error, sqlx_core::error::Error::WorkerCrashed));
    assert_eq!(unsafe { sqlite3_step(statement) }, SQLITE_ROW);
    checkpoint("busy_close_failed_and_native_statement_still_runs");
    assert_eq!(unsafe { sqlite3_finalize(statement) }, SQLITE_OK);
    assert_eq!(unsafe { sqlite3_close(database) }, SQLITE_OK);
    // The forgotten token is fail-closed until this owned process exits.
    checkpoint("manual_native_close_cannot_release_forgotten_custody");
}
