//! Lock-only access to an existing owner database.

#![expect(
    clippy::disallowed_methods,
    reason = "this is part of the centralized SQLite connection shim"
)]

use crate::SqliteConfig;
use log::LevelFilter;
use sqlx::ConnectOptions;
use sqlx::Error;
use sqlx::SqlSafeStr;
use sqlx::Sqlite;
use sqlx::SqlitePool;
use sqlx::Transaction;
use sqlx::sqlite::SqliteConnectOptions;
use sqlx::sqlite::SqlitePoolOptions;
use sqlx::sqlite::SqliteSynchronous;
use std::path::Path;
use std::time::Duration;

/// Holds the real SQLite writer lock while exposing a read-only transaction.
/// Connections permanently veto commits; the owner must retain the original
/// database identity/lifetime guard. This primitive supplies no Current witness.
#[derive(Clone)]
pub struct SqliteReadBarrier {
    pool: SqlitePool,
}

impl SqliteConfig {
    /// Open an existing owner database without creating a new writer generation.
    /// Callers must retain their original recovery/identity guard for its lifetime.
    pub async fn open_durable_read_barrier(&self, path: &Path) -> Result<SqliteReadBarrier, Error> {
        let options = SqliteConnectOptions::new()
            .filename(path)
            .create_if_missing(false)
            .synchronous(SqliteSynchronous::Full)
            .foreign_keys(true)
            .busy_timeout(Duration::from_secs(5))
            .pragma("query_only", "ON")
            .log_statements(LevelFilter::Off);
        let pool = SqlitePoolOptions::new()
            .max_connections(1)
            .after_connect(|connection, _metadata| {
                Box::pin(async move {
                    connection.lock_handle().await?.set_commit_hook(|| false);
                    Ok(())
                })
            })
            .connect_with(options)
            .await?;
        Ok(SqliteReadBarrier { pool })
    }
}

impl SqliteReadBarrier {
    /// Acquire the actual writer lock, then restrict the transaction to reads.
    ///
    /// SQLite may fall back to a read-only open; BEGIN IMMEDIATE on a read-only
    /// database does not exclude writers. Check the actual handle before briefly
    /// clearing query_only to acquire the lock. Closing on drop prevents any
    /// error or cancellation from returning the intermediate state to the pool.
    pub async fn begin(&self) -> Result<Transaction<'static, Sqlite>, Error> {
        let mut connection = self.pool.acquire().await?;
        connection.close_on_drop();
        {
            let mut handle = connection.lock_handle().await?;
            // SAFETY: the SQLx handle lock keeps the connection alive and excludes
            // worker access. The static NUL-terminated name identifies main.
            let read_only = unsafe {
                libsqlite3_sys::sqlite3_db_readonly(
                    handle.as_raw_handle().as_ptr(),
                    c"main".as_ptr(),
                )
            };
            if read_only != 0 {
                return Err(Error::Protocol(
                    "read barrier requires a writable existing main database".to_string(),
                ));
            }
        }
        sqlx::query("PRAGMA query_only=OFF")
            .execute(&mut *connection)
            .await?;
        let mut transaction =
            Transaction::begin(connection, Some("BEGIN IMMEDIATE".into_sql_str())).await?;
        sqlx::query("PRAGMA query_only=ON")
            .execute(&mut *transaction)
            .await?;
        Ok(transaction)
    }

    /// Drain all lock-only connections before releasing the owner's guard.
    pub async fn close(&self) {
        self.pool.close().await;
    }
}
