//! Bounded checkpoint wire validation. This module performs no I/O or admission.

use codex_hepta_contracts::Sha256Digest;
use sha2::Digest;
use sha2::Sha256;

use crate::ObjectiveSenderCheckpointError as Error;
use crate::cognitive_checkpoint_wire::COGNITIVE_CHECKPOINT_FRAME_LIMIT;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointCommand;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointHead;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointPhase;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointReceipt;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointRequest;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointResponse;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointRetainedProposal;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointRetainedRequest;
use crate::cognitive_checkpoint_wire::CognitiveCheckpointStatus;

impl CognitiveCheckpointRequest {
    /// Encode the exact bytes whose digest the protected peer must retain.
    /// The caller must send these bytes unchanged over its authenticated channel.
    /// This owner codec handles reads and publication transitions; it does not
    /// encode the protected service's FencePublisher lifecycle operation.
    pub fn encode(&self) -> Result<Vec<u8>, Error> {
        self.sequence()?;
        let valid_shape = match self.command {
            CognitiveCheckpointCommand::ReadHead => {
                self.expected.is_none() && self.proposal.is_none()
            }
            CognitiveCheckpointCommand::PreparePublication => {
                self.expected.is_some() && self.proposal.is_some()
            }
            CognitiveCheckpointCommand::BeginWrites
            | CognitiveCheckpointCommand::CommitPublication => {
                self.expected.is_some() && self.proposal.is_none()
            }
            CognitiveCheckpointCommand::FencePublisher => false,
        };
        if self.schema_version != 1
            || self.operation != "runtime.cognitive.checkpoint.v1"
            || !valid_shape
        {
            return Err(Error::Refused);
        }
        let mut digests = Vec::new();
        if let Some(expected) = &self.expected {
            digests.push(&expected.head_sha256);
        }
        if let Some(proposal) = &self.proposal {
            digests.extend([
                &proposal.cut.schema_digest,
                &proposal.cut.state_digest,
                &proposal.writer_fence.lease_sha256,
                &proposal.writer_fence.authority_grant_digest,
                &proposal.writer_fence.fencing_token_digest,
            ]);
        }
        if !valid_digests(digests) {
            return Err(Error::Refused);
        }
        let bytes = serde_json::to_vec(self).map_err(|_| Error::Unavailable)?;
        if bytes.len() > COGNITIVE_CHECKPOINT_FRAME_LIMIT {
            return Err(Error::Refused);
        }
        Ok(bytes)
    }

    /// Decode an ACK bound to this request without authenticating its sender.
    /// This checks framing and wire consistency only. The owner must additionally
    /// validate the full predecessor/successor, live authority and fresh Current
    /// before recovery or binding. This method performs no I/O or admission.
    pub fn decode_ack(&self, response: &[u8]) -> Result<CognitiveCheckpointReceipt, Error> {
        self.encode()?;
        if response.is_empty() || response.len() > COGNITIVE_CHECKPOINT_FRAME_LIMIT {
            return Err(Error::Unavailable);
        }
        let response: CognitiveCheckpointResponse =
            serde_json::from_slice(response).map_err(|_| Error::Unavailable)?;
        if response.schema_version != 1 || response.request_id != self.request_id {
            return Err(Error::Unavailable);
        }
        match response.status {
            CognitiveCheckpointStatus::Refused => return Err(Error::Refused),
            CognitiveCheckpointStatus::Unknown => return Err(Error::Unavailable),
            CognitiveCheckpointStatus::Ack => {}
        }
        if response.reason.is_some() {
            return Err(Error::Unavailable);
        }
        let receipt = CognitiveCheckpointReceipt {
            token: response.token.ok_or(Error::Unavailable)?,
            head: response.head.ok_or(Error::Unavailable)?,
        };
        validate_receipt(&receipt)?;
        Ok(receipt)
    }

    /// Validate a response against a caller-held predecessor.
    /// Mutation ACKs must describe the entire exact successor, including the
    /// digest of encode() bytes and the full predecessor token. ReadHead checks
    /// identity and monotonicity only: Dirty/Publishing/Fenced observations are
    /// not Current admission. No clock, authority, transport or grant is created.
    pub fn decode_against(
        &self,
        previous: &CognitiveCheckpointReceipt,
        response: &[u8],
    ) -> Result<CognitiveCheckpointReceipt, Error> {
        let request_bytes = self.encode()?;
        let (epoch, sequence) = self.sequence()?;
        validate_receipt(previous)?;
        if epoch != previous.head.publisher_epoch
            || sequence <= previous.head.last_mutation_sequence
            || (self.command != CognitiveCheckpointCommand::ReadHead
                && self.expected.as_ref() != Some(&previous.token))
        {
            return Err(Error::Refused);
        }
        let observed = self.decode_ack(response)?;
        if self.command == CognitiveCheckpointCommand::ReadHead {
            if observed.head.owner_agent_id != previous.head.owner_agent_id
                || observed.head.publisher_epoch != previous.head.publisher_epoch
                || observed.head.writer_binding != previous.head.writer_binding
                || observed.head.peer_lifetime != previous.head.peer_lifetime
                || observed.head.counter < previous.head.counter
                || observed.head.last_mutation_sequence < previous.head.last_mutation_sequence
            {
                return Err(Error::Refused);
            }
            return Ok(observed);
        }
        let mut expected = previous.head.clone();
        expected.counter = expected.counter.checked_add(1).ok_or(Error::Unavailable)?;
        expected.last_mutation_sequence = sequence;
        expected.last_request = Some(CognitiveCheckpointRetainedRequest {
            request_id: self.request_id.clone(),
            request_sha256: Sha256Digest::from_sha256_output(Sha256::digest(request_bytes)),
            command: self.command,
            predecessor_token: previous.token.clone(),
        });
        match (self.command, previous.head.phase) {
            (CognitiveCheckpointCommand::BeginWrites, CognitiveCheckpointPhase::Current)
                if previous.head.proposal.is_none() =>
            {
                expected.phase = CognitiveCheckpointPhase::Dirty;
            }
            (CognitiveCheckpointCommand::PreparePublication, CognitiveCheckpointPhase::Dirty)
                if previous.head.proposal.is_none() =>
            {
                let proposal = self.proposal.as_ref().ok_or(Error::Refused)?;
                if !proposal_matches(&previous.head, &proposal.cut, &proposal.writer_fence) {
                    return Err(Error::Refused);
                }
                expected.phase = CognitiveCheckpointPhase::Publishing;
                expected.proposal = Some(CognitiveCheckpointRetainedProposal {
                    request_id: self.request_id.clone(),
                    cut: proposal.cut.clone(),
                    writer_fence: proposal.writer_fence.clone(),
                });
            }
            (CognitiveCheckpointCommand::CommitPublication, CognitiveCheckpointPhase::Publishing) => {
                let proposal = previous.head.proposal.as_ref().ok_or(Error::Refused)?;
                if !proposal_matches(&previous.head, &proposal.cut, &proposal.writer_fence) {
                    return Err(Error::Refused);
                }
                expected.last_current = proposal.cut.clone();
                expected.phase = CognitiveCheckpointPhase::Current;
                expected.proposal = None;
            }
            _ => return Err(Error::Refused),
        }
        if observed.head != expected || observed.token.head_sha256 == previous.token.head_sha256 {
            return Err(Error::Unavailable);
        }
        Ok(observed)
    }

    fn sequence(&self) -> Result<(u64, u64), Error> {
        if self.request_id.len() != 32
            || !self.request_id.bytes().all(|byte| byte.is_ascii_hexdigit())
        {
            return Err(Error::Refused);
        }
        let epoch = u64::from_str_radix(&self.request_id[..16], 16).map_err(|_| Error::Refused)?;
        let sequence = u64::from_str_radix(&self.request_id[16..], 16).map_err(|_| Error::Refused)?;
        if epoch == 0 || sequence == 0 {
            return Err(Error::Refused);
        }
        Ok((epoch, sequence))
    }
}

fn validate_receipt(receipt: &CognitiveCheckpointReceipt) -> Result<(), Error> {
    if receipt.head.schema_version != 1
        || receipt.head.publisher_epoch == 0
        || receipt.head.counter == 0
        || receipt.token.publisher_epoch != receipt.head.publisher_epoch
        || receipt.token.counter != receipt.head.counter
        || receipt.head.last_current.owner_agent_id != receipt.head.owner_agent_id
        || receipt.head.last_current.profile != "hepta:cognitive:exact-current-cut:v1"
    {
        return Err(Error::Unavailable);
    }
    let mut digests = vec![
        &receipt.token.head_sha256,
        &receipt.head.writer_binding.authority_grant_digest,
        &receipt.head.writer_binding.fencing_token_digest,
        &receipt.head.last_current.schema_digest,
        &receipt.head.last_current.state_digest,
    ];
    if let Some(proposal) = &receipt.head.proposal {
        digests.extend([
            &proposal.cut.schema_digest,
            &proposal.cut.state_digest,
            &proposal.writer_fence.lease_sha256,
            &proposal.writer_fence.authority_grant_digest,
            &proposal.writer_fence.fencing_token_digest,
        ]);
    }
    if let Some(request) = &receipt.head.last_request {
        digests.extend([&request.request_sha256, &request.predecessor_token.head_sha256]);
    }
    if !valid_digests(digests) {
        return Err(Error::Unavailable);
    }
    Ok(())
}

fn proposal_matches(
    head: &CognitiveCheckpointHead,
    cut: &crate::CognitiveRecoveryAnchor,
    writer: &crate::ObjectiveSenderWriterFence,
) -> bool {
    let binding = &head.writer_binding;
    cut.owner_agent_id == head.owner_agent_id
        && cut.profile == head.last_current.profile
        && writer.owner_agent_id == head.owner_agent_id
        && binding.lease_id == writer.lease_id
        && binding.generation == writer.generation
        && binding.authority_epoch == writer.authority_epoch
        && binding.owner_epoch == writer.owner_epoch
        && binding.authority_grant_digest == writer.authority_grant_digest
        && binding.fencing_token_digest == writer.fencing_token_digest
        && binding.expires_at_unix_seconds == writer.expires_at_unix_seconds
}

fn valid_digests<'a>(digests: impl IntoIterator<Item = &'a Sha256Digest>) -> bool {
    digests
        .into_iter()
        .all(|digest| Sha256Digest::parse(digest.as_str()).is_ok())
}
