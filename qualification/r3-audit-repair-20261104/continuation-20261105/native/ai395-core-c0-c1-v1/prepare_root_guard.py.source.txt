from pathlib import Path
import ast
import datetime
import hashlib
import json

SOURCE = ROOT / "ci-audit/core-wait-span-20261105/ai395-cold-prepare-v1"
OUT = ROOT / "hepta-audit-fifth/inputs"


def identity(path):
    raw = path.read_bytes()
    return {"local_path": str(path), "bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest(),
            "git_blob": hashlib.sha1(f"blob {len(raw)}\0".encode() + raw).hexdigest()}


def read(path):
    return json.loads(path.read_bytes())


manifest = read(SOURCE / "manifest.json")
for key in ("runner", "request", "process_start", "initial_draft", "delta"):
    row = manifest[key]
    actual = identity(Path(row["path"]))
    assert all(actual[k] == row[k] for k in ("bytes", "sha256", "git_blob")), key
runner = Path(manifest["runner"]["path"]).read_text()
ast.parse(runner)
initial = Path(manifest["initial_draft"]["path"]).read_text()
assert runner == initial.replace("'PYTHONPYCACHEPREFIX': 'python-pycache'}.items():",
    "'PYTHONPYCACHEPREFIX': 'python-pycache', 'RUSTUP_HOME': 'rustup-home'}.items():")
request = read(Path(manifest["request"]["path"]))
process = read(Path(manifest["process_start"]["path"]))
assert request["execute"] is False and request["root_execution_guard"] is None
assert process["host"] == "ai395" and process["executable"] == "/usr/bin/python3"
assert process["args"] == ["-c", runner]
assert process["stdin"] == Path(manifest["request"]["path"]).read_text()
assert process["cwd"] == "/home/qq/work" and process["timeout_seconds"] == 21600
assert process["idempotency_key"] == request["idempotency_key"] == "hepta-ai395-core-cold-prepare-20261105-v1"
assert process["env"] == {"PYTHONDONTWRITEBYTECODE": "1", "GIT_OPTIONAL_LOCKS": "0"}
assert request["stage"] == "prepare"
assert request["readiness"]["sha256"] == "c61d68f27c06ab53c0b2befc36a3b6cc2b56184e8d77fa4ee77cece37761bfd3"
