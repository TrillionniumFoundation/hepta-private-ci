def pin(path: Path):
    body = path.read_bytes()
    return {"path": str(path.relative_to(ROOT)), "bytes": len(body),
            "sha256": hashlib.sha256(body).hexdigest(),
            "git_blob": hashlib.sha1(b"blob " + str(len(body)).encode() + b"\0" + body).hexdigest()}


def read(path):
    return json.loads((ROOT / path).read_text())


def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def source_anchor(path, needle, radius=0):
    lines = (SRC / path).read_text().splitlines()
    matches = [n for n, line in enumerate(lines, 1) if needle in line]
    assert len(matches) == 1, (path, needle, matches)
    n = matches[0]
    return {"path": path, "git_blob": leaf[path]["sha"], "line": n,
            "excerpt": "\n".join(lines[max(0, n - 1 - radius):n + radius])}
