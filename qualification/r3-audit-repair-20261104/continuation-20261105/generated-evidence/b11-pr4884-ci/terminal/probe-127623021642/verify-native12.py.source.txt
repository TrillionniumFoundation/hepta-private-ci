def load(path):
    return json.loads(path.read_bytes())


def digest(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(),
            'git_blob': hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()}


def unwrap(value):
    assert not value.get('isError', False)
    result = value.get('structuredContent') or json.loads(next(x['text'] for x in value['content'] if x['type'] == 'text'))
    return json.loads(result['content']) if isinstance(result.get('content'), str) else result


def summaries(text):
    return re.findall(r'^test result: (ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out; finished in ([0-9.]+)s$', text, re.M)


def pair(path, members, expected_error=None):
    raw, xml_raw = path.read_bytes(), path.with_suffix('.xml').read_bytes()
    text, xml = raw.decode(), ET.fromstring(xml_raw)
    assert len(xml.findall('.//testcase')) == 1
    assert [x.get('message') for x in xml.findall('.//error')] == ([] if expected_error is None else [expected_error])
    assert not xml.findall('.//skipped') and not xml.findall('.//failure')
    assert xml.find('.//system-out').text == text.replace('\r\n', '\n').replace('\r', '\n').replace('\x1b', '?')
    return text, {'member': path.relative_to(members).as_posix(), 'raw': digest(raw), 'XML': digest(xml_raw),
                  'original_ESC_count': text.count('\x1b'), 'original_CR_count': text.count('\r'),
                  'XML_entire_system_out_matches_after_only_XML_newline_and_ESC_serialization': True,
                  'XML_outer_case': xml.find('.//testcase').attrib, 'expected_error': expected_error}


def one_log(scope, kind):
    paths = list((scope / 'bazel-execution-logs' / kind).glob('invocation-*/targets/0000/test.log'))
    assert len(paths) == 1, (scope, kind, len(paths))
    return paths[0]


def reconstruct_archive():
    manifest = load(P / 'archive/manifest.json')
    assert manifest['artifact_id'] == ARTIFACT
    chunks, offset = [], 0
    for part in manifest['parts']:
        name = PurePosixPath(part['path'])
        assert not name.is_absolute() and '..' not in name.parts
        encoded = (P / name).read_bytes()
        assert digest(encoded) == part['encoded'] and encoded.endswith(b'\n')
        raw = base64.b64decode(encoded[:-1], validate=True)
        assert digest(raw) == part['raw'] and part['offset'] == offset
        offset += len(raw)
        chunks.append(raw)
    archive = b''.join(chunks)
    assert len(chunks) == 18 and digest(archive) == manifest['archive']
    assert len(archive) == 2302874
    assert digest(archive)['sha256'] == 'b21172467ad92e47439defa6075099cf7ba9fcf56b22221de145f0434bffe4f2'
    if (P / 'artifact.zip').exists():
        assert (P / 'artifact.zip').read_bytes() == archive
    return archive


def verify_archive(archive, members):
    claimed = load(P / 'artifact-verification.json')
    assert claimed['artifact_id'] == ARTIFACT and claimed['zip_bytes'] == len(archive)
    assert claimed['zip_sha256'] == digest(archive)['sha256']
    rows = {row['path']: row for row in claimed['members']}
    assert len(rows) == len(claimed['members']) == 84
    total = 0
    with zipfile.ZipFile(io.BytesIO(archive)) as z:
        infos = z.infolist()
        assert z.testzip() is None and len(infos) == len({i.filename for i in infos}) == 84
        assert set(rows) == {i.filename for i in infos}
        for info in infos:
            name = PurePosixPath(info.filename)
            assert not info.is_dir() and not name.is_absolute() and '..' not in name.parts
            assert '\\' not in info.filename and not stat.S_ISLNK(info.external_attr >> 16)
            raw = z.read(info.filename)
            row = rows[info.filename]
            assert {k: row[k] for k in digest(raw)} == digest(raw)
            assert row['crc32'] == f'{info.CRC:08x}'
            target = members / name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(raw)
            total += len(raw)
    assert total == 3000850
    return {'member_count': 84, 'uncompressed_bytes': total, 'all_original_CRC_and_bytes_checked': True,
            'computed_member_git_blob_ids_are_not_claimed_as_separate_actual_Git_objects': True}


def verify_merkle():
    proof = load(P / 'source-merkle-proof.json')
    assert proof['tree'] == TREE
    nodes = {n['directory']: n for n in proof['nodes']}
    assert len(nodes) == len(proof['nodes']) == 18
    for directory, node in nodes.items():
        entries = node['entries']
        assert len(entries) == len({e['name'] for e in entries})
        assert all('/' not in e['name'] and '\0' not in e['name'] for e in entries)
        ordered = sorted(entries, key=lambda e: (e['name'] + ('/' if e['type'] == 'tree' else '')).encode())
        assert entries == ordered
        raw = b''.join((('40000' if e['type'] == 'tree' else e['mode']) + ' ' + e['name']).encode() + b'\0' + bytes.fromhex(e['sha']) for e in entries)
        assert hashlib.sha1(b'tree ' + str(len(raw)).encode() + b'\0' + raw).hexdigest() == node['sha']
        if directory:
            parent, _, name = directory.rpartition('/')
            match, = [e for e in nodes[parent]['entries'] if e['name'] == name]
            assert match['type'] == 'tree' and match['mode'] == '040000' and match['sha'] == node['sha']
    assert nodes['']['sha'] == TREE
    actual = load(P / 'actual-tree-root.original.json')['response']
    assert actual['sha'] == TREE and not actual.get('truncated', False)
    original_entries = {e['path']: {k: e[k] for k in ['mode', 'type', 'sha']} for e in actual['tree']}
    assert original_entries == {e['name']: {k: e[k] for k in ['mode', 'type', 'sha']} for e in nodes['']['entries']}
    leaves = {}
    for path in proof['selected_leaves']:
        parent, _, name = path.rpartition('/')
        entry, = [e for e in nodes[parent]['entries'] if e['name'] == name]
        assert entry['type'] == 'blob'
        leaves[path] = {k: entry[k] for k in ['mode', 'type', 'sha']}
    assert len(leaves) == len(proof['selected_leaves']) == 83
    return leaves
