def load(path):
    return json.loads(path.read_bytes())


def digest(raw):
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest(),
            'git_blob': hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()}


def unwrap(value):
    assert not value.get('isError', False)
    result = value.get('structuredContent') or json.loads(next(x['text'] for x in value['content'] if x['type'] == 'text'))
    return json.loads(result['content']) if isinstance(result.get('content'), str) else result


def summaries(text):
    return re.findall(r'^test result: (ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out; finished in ([0-9.]+)s$', text, re.M)


def pair(path, members, expected_error=None):
    raw, xml_raw = path.read_bytes(), path.with_suffix('.xml').read_bytes()
    text, xml = raw.decode(), ET.fromstring(xml_raw)
    assert len(xml.findall('.//testcase')) == 1
    assert [x.get('message') for x in xml.findall('.//error')] == ([] if expected_error is None else [expected_error])
    assert not xml.findall('.//skipped') and not xml.findall('.//failure')
    assert xml.find('.//system-out').text == text.replace('\r\n', '\n').replace('\r', '\n').replace('\x1b', '?')
    return text, {'member': path.relative_to(members).as_posix(), 'raw': digest(raw), 'XML': digest(xml_raw),
                  'original_ESC_count': text.count('\x1b'), 'original_CR_count': text.count('\r'),
                  'XML_entire_system_out_matches_after_only_XML_newline_and_ESC_serialization': True,
                  'XML_outer_case': xml.find('.//testcase').attrib, 'expected_error': expected_error}


def one_log(scope, kind):

def reconstruct_archive():
    manifest = load(P / 'archive/manifest.json')
    assert manifest['artifact_id'] == ARTIFACT
    chunks, offset = [], 0
    for part in manifest['parts']:
        name = PurePosixPath(part['path'])
        assert not name.is_absolute() and '..' not in name.parts
        encoded = (P / name).read_bytes()
        assert digest(encoded) == part['encoded'] and encoded.endswith(b'\n')
        raw = base64.b64decode(encoded[:-1], validate=True)
        assert digest(raw) == part['raw'] and part['offset'] == offset
        offset += len(raw)
        chunks.append(raw)
    archive = b''.join(chunks)
    assert len(chunks) == 18 and digest(archive) == manifest['archive']
    assert len(archive) == 2302073
    assert digest(archive)['sha256'] == 'b67aa17b107136aa1e934065e671e35b99c1a8b74a82ad7edf40f21c58128ca8'
    if (P / 'artifact.zip').exists():
        assert (P / 'artifact.zip').read_bytes() == archive
    return archive


def verify_archive(archive, members):
    claimed = load(P / 'artifact-verification.json')
    assert claimed['artifact_id'] == ARTIFACT and claimed['zip_bytes'] == len(archive)
    assert claimed['zip_sha256'] == digest(archive)['sha256']
    rows = {row['path']: row for row in claimed['members']}
    assert len(rows) == len(claimed['members']) == 84
    total = 0
    with zipfile.ZipFile(io.BytesIO(archive)) as z:
        infos = z.infolist()
        assert z.testzip() is None and len(infos) == len({i.filename for i in infos}) == 84
        assert set(rows) == {i.filename for i in infos}
        for info in infos:
            name = PurePosixPath(info.filename)
            assert not info.is_dir() and not name.is_absolute() and '..' not in name.parts
            assert '\\' not in info.filename and not stat.S_ISLNK(info.external_attr >> 16)
            raw = z.read(info.filename)
            row = rows[info.filename]
            assert {k: row[k] for k in digest(raw)} == digest(raw)
            assert row['crc32'] == f'{info.CRC:08x}'
            target = members / name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(raw)
            total += len(raw)
    assert total == 3011303
    return {'member_count': 84, 'uncompressed_bytes': total, 'all_original_CRC_and_bytes_checked': True,
            'computed_member_git_blob_ids_are_not_claimed_as_separate_actual_Git_objects': True}


def verify_merkle():
    proof = load(P / 'source-merkle-proof.json')
    assert proof['tree'] == TREE
    nodes = {n['directory']: n for n in proof['nodes']}
    assert len(nodes) == len(proof['nodes']) == 18
    for directory, node in nodes.items():
        entries = node['entries']
        assert len(entries) == len({e['name'] for e in entries})
        assert all('/' not in e['name'] and '\0' not in e['name'] for e in entries)
        ordered = sorted(entries, key=lambda e: (e['name'] + ('/' if e['type'] == 'tree' else '')).encode())
        assert entries == ordered
        raw = b''.join((('40000' if e['type'] == 'tree' else e['mode']) + ' ' + e['name']).encode() + b'\0' + bytes.fromhex(e['sha']) for e in entries)
        assert hashlib.sha1(b'tree ' + str(len(raw)).encode() + b'\0' + raw).hexdigest() == node['sha']
        if directory:
            parent, _, name = directory.rpartition('/')
            match, = [e for e in nodes[parent]['entries'] if e['name'] == name]
            assert match['type'] == 'tree' and match['mode'] == '040000' and match['sha'] == node['sha']
    assert nodes['']['sha'] == TREE
    actual = load(P / 'actual-tree-root.original.json')['response']
    assert actual['sha'] == TREE and not actual.get('truncated', False)
    original_entries = {e['path']: {k: e[k] for k in ['mode', 'type', 'sha']} for e in actual['tree']}
    assert original_entries == {e['name']: {k: e[k] for k in ['mode', 'type', 'sha']} for e in nodes['']['entries']}
    leaves = {}
    for path in proof['selected_leaves']:
        parent, _, name = path.rpartition('/')
        entry, = [e for e in nodes[parent]['entries'] if e['name'] == name]
        assert entry['type'] == 'blob'
        leaves[path] = {k: entry[k] for k in ['mode', 'type', 'sha']}
    assert len(leaves) == len(proof['selected_leaves']) == 83
    return leaves


def rust_observations(whole):
    prefix = 'PROBE_OBSERVATION_ONLY neither_privileges '
    records = []
    for line_number, line in enumerate(whole.splitlines(), 1):
        if prefix not in line:
            continue
        phase, raw = line.split(prefix, 1)[1].split(' ', 1)
        obj = json.loads(raw)  # No recovery or repair of a malformed/interleaved line.
        records.append({'phase': phase, 'line': line_number, 'physical_line_utf8': digest(line.encode()), 'value': obj})
    assert [r['phase'] for r in records] == ['filesystem', 'subjects', 'rust_before', 'rust_physical', 'rust_before', 'rust_physical', 'rust_before', 'rust_physical']
    filesystem = records[0]['value']
    assert filesystem['filesystem'] == 'NTFS'
    subjects = records[1]['value']
    ordinary, disabled = subjects['ordinary'], subjects['disable_max_privilege']
    assert ordinary['identity'] == disabled['identity']
    assert ordinary['identity']['restricted_sids'] == []
    assert ordinary['token_type'] == disabled['token_type'] == 1
    assert ordinary['impersonation_level'] is None and disabled['impersonation_level'] is None
    ordinary_enabled = [v['name'] for v in ordinary['privileges'] if v['enabled']]
    disabled_enabled = [v['name'] for v in disabled['privileges'] if v['enabled']]
    assert ordinary_enabled == ['SeBackupPrivilege', 'SeRestorePrivilege', 'SeDebugPrivilege', 'SeChangeNotifyPrivilege', 'SeImpersonatePrivilege', 'SeCreateGlobalPrivilege']
    assert disabled_enabled == ['SeChangeNotifyPrivilege']
    names = ['ordinary-direct', 'ordinary-impersonated', 'disabled-impersonated']
    routes = ['Neither', 'TargetOnly', 'ParentOnly', 'Both']
    for i, name in enumerate(names):
        before, physical = records[2 + 2 * i]['value'], records[3 + 2 * i]['value']
        assert before['subject'] == physical['subject'] == name
        token = before['token']
        assert token['identity'] == ordinary['identity']
        assert before['is_thread'] is (i != 0)
        assert (token['token_type'], token['impersonation_level']) == ((1, None) if i == 0 else (2, 2))
        assert token['privileges'] == (disabled if i == 2 else ordinary)['privileges']
        assert len(physical['operations']) == len(physical['outcomes']) == 4
        for j, (operation, outcome) in enumerate(zip(physical['operations'], physical['outcomes'])):
            assert outcome['route'] == routes[j]
            assert operation['read'] == 'matrix-original' and operation['read_error'] is None
            denied = i == 2 and j == 0
            if denied:
                assert operation['file_error']['code'] == operation['directory_error']['code'] == 5
            else:
                assert operation['file_error'] is None and operation['directory_error'] is None
            assert outcome['file_deleted'] is (not denied) and outcome['directory_deleted'] is (not denied)
    pattern = re.compile(r'^delete_matrix privilege-observation-rust-(ordinary-direct|ordinary-impersonated|disabled-impersonated) path=("(?:[^"\\]|\\.)*") identity=\((\d+), (\d+), (\d+)\) directory=(\d+) pending=(\d+) links=(\d+) acl=\[([0-9a-f, ]+)\]$', re.M)
    acl_rows, per_subject = [], {n: {} for n in names}
    for match in pattern.finditer(whole):
        name = match[1]
        path = json.loads(match[2])
        relative = PureWindowsPath(path).relative_to(PureWindowsPath(filesystem['root']) / ('rust-' + name))
        key = relative.as_posix()
        assert key not in per_subject[name]
        acl = bytes(int(part.strip(), 16) for part in match[9].split(','))
        row = {'subject': name, 'relative': key, 'identity': [int(match[k]) for k in [3, 4, 5]],
               'directory': int(match[6]), 'pending': int(match[7]), 'links': int(match[8]),
               'ACL_hex': acl.hex(), 'line': whole.count('\n', 0, match.start()) + 1}
        assert row['identity'][0] == filesystem['serial']
        per_subject[name][key] = row
        acl_rows.append(row)
    assert len(acl_rows) == len({tuple(r['identity']) for r in acl_rows}) == 36
    expected_keys = {r + suffix for r in routes for suffix in ['', '/file', '/directory']}
    assert all(set(per_subject[name]) == expected_keys for name in names)
    for key in expected_keys:
        assert len({per_subject[name][key]['ACL_hex'] for name in names}) == 1
    for name in names:
        for suffix in ['/file', '/directory']:
            row = per_subject[name]['Neither' + suffix]
            assert (row['pending'], row['links']) == ((0, 1) if name == 'disabled-impersonated' else (1, 0))
    child_markers = {phase: sum(prefix + phase + ' ' in line for line in whole.splitlines())
                     for phase in ['child_before_release', 'child_finished', 'cmd_physical', 'complete']}
    assert all(count == 0 for count in child_markers.values())
    assert whole.count('Error: GetTokenInformation(TokenRestrictedSids) size query failed: 122') == 1
    return {'complete_raw_JSON_records': records, 'parse_repairs': 0, 'full_identity_fields_equal': True,
            'ordinary_enabled_privileges': ordinary_enabled, 'disabled_enabled_privileges': disabled_enabled,
            'actual_thread_token_pairs_type_2_level_2': True, 'complete_retained_object_rows': acl_rows,
            'three_separate_cohorts_have_equal_actual_ACL_bytes_and_distinct_retained_identities': True,
            'Rust_Neither_outcome_changes_with_all_privilege_filter_intervention': True,
            'single_SeRestore_causality_demonstrated': False, 'cmd_phase_markers': child_markers,
            'cmd_pair_and_owned_Job_cleanup_qualified': False,
            'cmd_stop': 'Existing process backend rejects the unrestricted-token launch precondition; Error 122 occurs before child creation. No cmd result is inferred.',
            'limits': ['Different ACL-equivalent fixtures, not reuse of one object across subjects.',
                       'Only three Rust cohorts completed; full observation test failed.',
                       'All-privilege filtering does not isolate any one privilege.',
                       'Strict sandbox outside-effects assertions remain failures.']}
