def identity(body):
    return {"bytes": len(body), "sha256": hashlib.sha256(body).hexdigest(),
            "git_blob": hashlib.sha1(b"blob " + str(len(body)).encode() + b"\0" + body).hexdigest()}


def record(path):
    return {"local_path": str(path), **identity(path.read_bytes())}


def load(path):
    return json.loads(path.read_bytes())


def write(name, value):
    path = DIRECTORY / name
    path.write_bytes((json.dumps(value, indent=2, ensure_ascii=False) + "\n").encode())
    return record(path)


def unwrap(envelope):
    assert not envelope.get("isError", False)
    value = envelope.get("structuredContent") or json.loads(next(x["text"] for x in envelope["content"] if x["type"] == "text"))
    return json.loads(value["content"])


def xml_pair(path, expected_error=None):
    raw = path.read_bytes()
    xml = ET.fromstring(path.with_suffix(".xml").read_bytes())
    assert len(xml.findall(".//testcase")) == 1
    assert [x.get("message") for x in xml.findall(".//error")] == ([] if expected_error is None else [expected_error])
    assert not xml.findall(".//failure") and not xml.findall(".//skipped")
    assert b"\r" not in raw
    assert xml.find(".//system-out").text.encode() == raw.replace(b"\x1b", b"?")
    return {"raw": record(path), "XML": record(path.with_suffix(".xml")),
            "original_ESC_count": raw.count(b"\x1b"), "original_CR_count": 0,
            "whole_XML_system_out_equal_except_ESC_to_question_mark": True,
            "outer_case": xml.find(".//testcase").attrib, "expected_error": expected_error}
