def identity(body):
    return {'bytes':len(body),'sha256':hashlib.sha256(body).hexdigest(),
            'git_blob':hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest()}

def save(path, body):
    path.parent.mkdir(parents=True,exist_ok=True)
    if path.exists():
        assert path.read_bytes()==body,str(path)
    else:
        path.write_bytes(body)

def write(path, value):
    save(path,(json.dumps(value,ensure_ascii=False,indent=2)+'\n').encode())

def unwrap(value):
    assert not value.get('isError',False)
    obj=value.get('structuredContent') or json.loads(next(x['text'] for x in value['content'] if x['type']=='text'))
    return json.loads(obj['content']) if isinstance(obj.get('content'),str) else obj
