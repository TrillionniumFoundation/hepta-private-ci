def identity(path):
    data = path.read_bytes()
    return {"path": str(path.relative_to(ROOT)), "bytes": len(data),
            "sha256": hashlib.sha256(data).hexdigest(),
            "git_blob": hashlib.sha1(b"blob " + str(len(data)).encode() + b"\0" + data).hexdigest()}

def document(path):
    return json.loads(path.read_bytes())

def api(name):
    return json.loads(document(HERE / (name + ".original.json"))["response"]["structuredContent"]["content"])

def native_dto(path):
    obj = document(path)
    return json.loads(next(x["text"] for x in obj["response"]["content"] if x["type"] == "text"))

def cleaned(path):
    return re.sub(r"\x1b\[[0-?]*[ -/]*[@-~]", "", path.read_text())

def selected_lines(path, needles):
    text = cleaned(path)
    return [{"line": i, "text": line} for i, line in enumerate(text.splitlines(), 1)
            if any(needle in line for needle in needles)]
