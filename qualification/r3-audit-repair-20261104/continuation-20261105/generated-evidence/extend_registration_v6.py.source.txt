def add(path, destination, role, pin=None):
    path = Path(path)
    if not path.is_absolute():
        path = ROOT / path
    data = path.read_bytes()
    actual = {"bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(),
              "blob": hashlib.sha1(f"blob {len(data)}\0".encode() + data).hexdigest()}
    destination = destination if destination.startswith(PREFIX + "/") else PREFIX + "/" + destination
    assert destination not in items, destination
    row = {"repository_path": destination, "local_file": str(path), "mode": "100644", **actual, "role": role}
    if pin:
        assert actual["blob"] == pin.get("blob", pin.get("git_blob"))
        for key in ("bytes", "sha256"):
            if key in pin:
                assert actual[key] == pin[key], path
        row.update(actual_object_verified=True,
                   verification="Actual Git create/full byte-equal GET; root rehashed local original and reviewed its scope")
        row["prior_object_verification"] = pin
    items[destination] = row
