from pathlib import Path
import hashlib
import json

ROOT = Path(__file__).resolve().parent.parent

PREFIX = "qualification/r3-audit-repair-20261104/continuation-20261105"
plan = json.loads((BASE / "registration-plan-v6-extended-verified.json").read_bytes())
items = {r["repository_path"]:r for r in plan["items"]}
assert len(items) == 224 and all(r["actual_object_verified"] for r in items.values())

def add(path, destination, role, blob=None, verified=False):
    path = Path(path)
    if not path.is_absolute(): path = ROOT / path
    raw = path.read_bytes()
    actual = {"bytes":len(raw),"sha256":hashlib.sha256(raw).hexdigest(),
              "blob":hashlib.sha1(f"blob {len(raw)}\0".encode()+raw).hexdigest()}
    if blob: assert blob == actual["blob"], (path,blob,actual)
    dest = destination if destination.startswith(PREFIX+"/") else PREFIX+"/"+destination
    assert dest not in items, dest
    items[dest] = {"repository_path":dest,"local_file":str(path),"mode":"100644",**actual,
                   "role":role,"actual_object_verified":verified}
    if verified: items[dest]["verification"]="Actual Git create and full byte-equal GET; root rehashed local bytes and reviewed exact scope"
