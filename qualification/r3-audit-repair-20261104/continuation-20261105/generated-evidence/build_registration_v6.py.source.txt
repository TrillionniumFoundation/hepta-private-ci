"""Prepare an additive evidence-only V6 plan; never mutate Git or V5."""
from pathlib import Path
import hashlib
import json

ROOT = Path(__file__).resolve().parent.parent
OUT = ROOT / "hepta-audit-fifth"
PREFIX = "qualification/r3-audit-repair-20261104/continuation-20261105"
items = {}


def load(path):
    return json.loads((ROOT / path).read_bytes())


def add(local, relative, role, pin=None):
    path = Path(local)
    path = path if path.is_absolute() else ROOT / path
    raw = path.read_bytes()
    identity = {"bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest(),
                "blob": hashlib.sha1(f"blob {len(raw)}\0".encode() + raw).hexdigest()}
    destination = relative if relative.startswith(PREFIX + "/") else PREFIX + "/" + relative
    assert destination not in items, destination
    row = {"repository_path": destination, "local_file": str(path), "mode": "100644", **identity, "role": role}
    if pin is not None:
        assert identity["blob"] == pin.get("blob", pin.get("git_blob")), path
        for key in ("bytes", "sha256"):
            if key in pin:
                assert identity[key] == pin[key], path
        row["prior_object_verification"] = pin
        row["object_verified_by"] = "Actual Git create/full GET, with root local bytes revalidated"
    items[destination] = row
