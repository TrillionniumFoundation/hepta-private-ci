checks=[]
def check(name, condition, detail=None):
    checks.append({'name': name, 'passed': bool(condition), 'detail': detail})
    if not condition:
        raise AssertionError((name, detail))

def rows(name):
    with gzip.open(RUN / name, 'rt') as f:
        return [json.loads(line) for line in f]

def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

started=time.perf_counter_ns()
manifest=json.loads((RUN/'FROZEN_MANIFEST.json').read_text())
result=json.loads((RUN/'RESULT.json').read_text())
receipt=json.loads((ROOT/'qualification/act2-learning-20261110/execution-run2-receipt.json').read_text())
manifest_hashes=json.loads((RUN/'SHA256SUMS.json').read_text())
check('recorded_source_before_equals_after',receipt['source_before']==receipt['source_after'])
for name,h in manifest['source_files_sha256'].items():
    check('source_hash_'+name,digest(SOURCE/name)==h==receipt['source_before'][name])
for name,h in manifest_hashes.items():
    check('artifact_hash_'+name,digest(RUN/name)==h)

    left={t[field] for t in train_tasks.values()};right={t[field] for t in confirm_tasks.values()}
    check('split_disjoint_'+field,not left&right,{'train':sorted(left),'confirmation':sorted(right)})

# Inspect retained actual SQLite bytes through a new read-only in-memory observer.
schemas=[('items','item_key','value'),('objects','object_key','content'),('documents','doc_id','body'),('records','record_id','payload')]
with zipfile.ZipFile(RUN/'destination-databases.zip') as archive:
    check('all_3280_destination_databases_retained',len(archive.namelist())==3280)
    for i,row in enumerate(train+confirm):
        task=(train_tasks if i<len(train) else confirm_tasks)[row['task_id']]
        data=archive.read(row['database_archive_member'])
        assert hashlib.sha256(data).hexdigest()==row['database_sha256']
        db=sqlite3.connect(':memory:')
        db.deserialize(data)
        db.execute('PRAGMA query_only=ON')
        table,key,value=schemas[task['schema']]
        actual=dict(db.execute(f'SELECT {key},{value} FROM {table}'))
        effects=list(db.execute('SELECT key,value FROM effects'))
        applied=list(db.execute('SELECT operation,payload FROM applied'))
        db.close()
        complete=all(actual[str(j)]==g for j,g in enumerate(task['goals']))
        partial=any(actual[str(j)]==g for j,g in enumerate(task['goals']))
        duplicate=len(effects)-len({k for k,v in effects})
        assert (complete,partial,duplicate)==(row['complete'],row['partial'],row['duplicate_effects'])
        assert len(effects)==len(applied)==row['applied_effects']
        for operation,payload in applied:
            index=int(operation.rsplit(':',1)[1])
            assert operation==task['id']+':'+str(index)
            assert payload==hashlib.sha256(task['goals'][index].encode()).hexdigest()
    check('all_destination_database_bytes_and_goal_scores',True)


# Re-execute frozen training and confirmation in fresh real SQLite databases.
# Timings are deliberately excluded from comparison; source-run originals remain untouched.
stable=['task_id','arm','destination','schema','composition','fault','complete','partial','duplicate_effects','claimed_success','unknown','goal_label_available','right_censored','primitive_calls','readback_queries','write_transactions','independent_oracle_queries','applied_effects','lost_replies','database_sha256','policy_decisions','private_state_bytes','decisions']
traces=[]
with tempfile.TemporaryDirectory(prefix='act2-independent-replay-') as temp:
    temp=Path(temp)
    for i,r in enumerate(train):
        fresh,trace=runner.episode(train_tasks[r['task_id']],'B1_macro',None,temp/f'train-{i}',True)
        check('training_replay_'+str(i),all(fresh[k]==r[k] for k in stable))
        traces.append(trace)
    # Construct arrays independently instead of using runner.arrays.
    width=max(len(f) for f,a in traces)
    x=np.zeros((len(traces),width,10));y=np.zeros((len(traces),width,3));mask=np.zeros((len(traces),width))
    for i,(features,actions) in enumerate(traces):
        x[i,:len(features)]=features
        y[i,:len(actions)]=np.eye(3)[actions]
        mask[i,:len(actions)]=1
    tensor_hash=hashlib.sha256(x.tobytes()+y.tobytes()+mask.tobytes()).hexdigest()
    check('all_fit_training_tensor_hashes',all(f['train_trajectory_sha256']==tensor_hash for f in fits))
    parameters_changed={}
    models={}
    for fit in fits:
        kind='head' if fit['arm']=='B3_head' else 'gru'
        model=Model(kind,fit['seed'])
        initial={k:v.copy() for k,v in model.p.items()}
        losses=model.fit(x,y,mask,epochs=220)
        saved_path=RUN/f"{fit['arm']}-seed-{fit['seed']}.npz"
        saved=np.load(saved_path)
        check('model_hash_'+saved_path.name,digest(saved_path)==fit['weights_sha256'])
        check('fit_steps_'+saved_path.name,len(losses)==fit['optimizer_steps']==220)
        check('fit_exact_losses_'+saved_path.name,losses==fit['losses'])
        check('fit_exact_parameters_'+saved_path.name,set(saved.files)==set(model.p) and all(np.array_equal(saved[k],model.p[k]) for k in model.p))
        changed=sum(not np.array_equal(initial[k],model.p[k]) for k in initial)
        check('fit_changed_parameters_'+saved_path.name,changed>0,{'changed_groups':changed,'total_groups':len(initial)})
        parameters_changed[saved_path.name]=changed
        models[(fit['seed'],fit['arm'])]=model
    mismatches=[]
    for i,r in enumerate(confirm):
        model=models.get((r['training_seed'],r['arm']))
        fresh,_=runner.episode(confirm_tasks[r['task_id']],r['arm'],model,temp/f'confirm-{i}',False)
        bad=[k for k in stable if fresh[k]!=r[k]]
        event_old=[{k:v for k,v in e.items() if k!='at_ns'} for e in r['events']]
        event_new=[{k:v for k,v in e.items() if k!='at_ns'} for e in fresh['events']]
        if bad or event_old!=event_new:
            mismatches.append({'episode':i,'fields':bad,'events_equal':event_old==event_new})
    check('all_3200_confirmation_replays_stable',not mismatches,{'mismatches':mismatches[:10]})
