def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def digest(value):
    return sha(json.dumps(value, sort_keys=True, ensure_ascii=False,
                          separators=(",", ":"), allow_nan=False).encode())


def no_duplicates(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("duplicate JSON key: " + key)
        result[key] = value
    return result


def parse(raw):
    def invalid(value):
        raise ValueError("nonfinite JSON constant: " + value)
    return json.loads(raw, object_pairs_hook=no_duplicates, parse_constant=invalid)


class Audit:
    def __init__(self):
        self.checks, self.files = [], {}

    def check(self, name, passed, **detail):
        self.checks.append({"name": name, "passed": bool(passed), **detail})
        return bool(passed)

    def read(self, path, limit=64 * 1024 * 1024):
        path = Path(path).absolute()
        before = path.lstat()
        if not stat.S_ISREG(before.st_mode) or not 0 < before.st_size <= limit:
            raise ValueError("bounded regular audit input required: " + str(path))
        with path.open("rb") as stream:
            raw = stream.read(limit + 1)
        after = path.stat()
        if len(raw) != before.st_size or (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
            raise ValueError("audit input changed while reading: " + str(path))
        entry = {"sha256": sha(raw), "bytes": len(raw), "mtime_ns": before.st_mtime_ns}
        if str(path) in self.files and self.files[str(path)] != entry:
            raise ValueError("audit input changed between reads: " + str(path))
        self.files[str(path)] = entry
        return raw

    def json(self, path):
        return parse(self.read(path, 4 * 1024 * 1024))

    def inventory(self, directory):
        result = {}
        for path in sorted(Path(directory).rglob("*")):
            if path.is_file() and ".cache" not in path.relative_to(directory).parts:
                raw = self.read(path)
                result[str(path.relative_to(directory))] = {"sha256": sha(raw), "bytes": len(raw)}
        if not 1 <= len(result) <= 64:
            raise ValueError("checkpoint inventory bound")
        return result

    def checkpoint(self, directory, expected, label):
        directory = Path(directory)
        raw = self.read(directory / "lineage.json")
        manifest = parse(raw)
        config = self.read(directory / "adapter_config.json")
        tensor = self.read(directory / "adapter_model.safetensors")
        self.check(label + ": manifest/config/tensor pins", sha(raw) == expected and sha(config) == manifest["config_sha256"] and sha(tensor) == manifest["adapter_sha256"])
        header_len = int.from_bytes(tensor[:8], "little")
        if not 1 <= header_len <= 1024 * 1024 or 8 + header_len >= len(tensor):
            raise ValueError("safetensors header bound")
        header, payload = parse(tensor[8:8 + header_len]), memoryview(tensor)[8 + header_len:]
        layout, ranges, count, total = {}, [], 0, 0
        state, raw_state = hashlib.sha256(), hashlib.sha256()
        for name, metadata in sorted(header.items()):
            if name == "__metadata__":
                continue
            width, dtype = WIDTHS[metadata["dtype"]]
            shape, (start, end) = metadata["shape"], metadata["data_offsets"]
            if any(type(v) is not int or v < 1 for v in shape) or end - start != math.prod(shape) * width or not 0 <= start < end <= len(payload):
                raise ValueError("invalid tensor shape/offset")
            value = payload[start:end]
            layout[name] = {"dtype": dtype, "shape": shape}
            ranges.append((start, end))
            count, total = count + math.prod(shape), total + end - start
            state.update(digest([name, shape, dtype]).encode())
            state.update(value)
            raw_state.update(name.encode())
            raw_state.update(value)
        cursor = 0
        for start, end in sorted(ranges):
            if start != cursor:
                raise ValueError("tensor payload gaps/overlap")
            cursor = end
        training = manifest["training"]
        self.check(label + ": actual tensor layout and private bytes", layout == manifest["tensor_layout"] and cursor == len(payload) == total and training["parameter_bytes"] == total and training["trainable_parameters"] == count)
        self.check(label + ": actual adapter state digest", state.hexdigest() == training["adapter_state_sha256"])
        return {"directory": str(directory.resolve()), "manifest": manifest,
                "inventory": self.inventory(directory), "parameter_bytes": total,
                "parameters": count, "adapter_state_sha256": state.hexdigest(),
                "raw_tensor_state_sha256": raw_state.hexdigest()}


def score(rows, facts):
    if len(rows) != len(facts):
        raise ValueError("incomplete journal")
    correct = sum(row["text"] == fact["value"] for row, fact in zip(rows, facts, strict=True))
    return {"correct": correct, "total": len(facts), "exact_match": correct / len(facts)}

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--run", type=Path, required=True)
    parser.add_argument("--repo", type=Path, required=True)
    parser.add_argument("--witness", type=Path, required=True)
    parser.add_argument("--parent-witness", type=Path, required=True)
