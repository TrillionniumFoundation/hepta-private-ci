def digest(data):
    return hashlib.sha256(data).hexdigest()


def save(name, value):
    with (OUT/name).open('x') as target:
        json.dump(value, target, indent=2, sort_keys=True, allow_nan=False)
        target.write('\n')


original_manifest_bytes = (FORMAL/'EVIDENCE_MANIFEST.json').read_bytes()
manifest = json.loads(original_manifest_bytes)
summary = json.loads((FORMAL/'SUMMARY.json').read_bytes())
process = json.loads((PROCESS/'PROCESS.json').read_bytes())
assert process['returncode'] == 0 and summary['qualified']
assert summary['actual_sessions'] == 5120 and summary['actual_rollouts'] == 25600
assert summary['source_unchanged'] and not summary['failures'] and not summary['null_failures']
assert digest((PROCESS/'OUTPUT.log').read_bytes()) == process['output_sha256']

expected = {'formal_confirmation01/'+relative: value for relative, value in manifest['artifacts'].items()}
expected['formal_confirmation01/EVIDENCE_MANIFEST.json'] = digest(original_manifest_bytes)
for path in PROCESS.iterdir():
    if path.is_file():
        expected['process01/'+path.name] = digest(path.read_bytes())
covered, archives = set(), []


def make_archive(name, paths):
    target = OUT/name
    members = {}
    with target.open('xb') as raw:
        with gzip.GzipFile(fileobj=raw, mode='wb', filename='', mtime=0) as zipped:
            with tarfile.open(fileobj=zipped, mode='w|') as tar:
                for path in sorted(paths):
                    relative = str(path.relative_to(RAM))
                    data = path.read_bytes()
                    actual = digest(data)
                    assert actual == expected[relative], relative
                    assert relative not in covered, relative
                    members[relative] = actual
                    info = tar.gettarinfo(str(path), arcname=relative)
                    info.uid = info.gid = 0
                    info.uname = info.gname = ''
                    info.mtime = 0
                    tar.addfile(info, io.BytesIO(data))
    reconstructed = {}
    with tarfile.open(target, 'r|gz') as tar:
        for member in tar:
            assert member.isfile()
            reconstructed[member.name] = digest(tar.extractfile(member).read())
    assert reconstructed == members
    covered.update(members)
    record = dict(path=name, bytes=target.stat().st_size, sha256=digest(target.read_bytes()),
                  file_members=len(members), all_members_match_original_manifest=True,
                  stream_roundtrip_verified=True)
    assert record['bytes'] <= 5*1024*1024, record
    archives.append(record)
    print(json.dumps(record), flush=True)

def semantic(result):
    events = copy.deepcopy(result['events'])
    for event in events:
        if 'decoded' in event:
            event['decoded'] = {key:event['decoded'][key] for key in
                                ('observation', 'logits', 'payload_sha256', 'payload_bytes')}
    return dict(score=result['score'], counts=result['counts'], events=events,
                elapsed=result['task_elapsed'])


identifiers, seeds = set(), set()
decoder_receipts, generator_files = 0, 0
ordinary_equal, hard_qvi_equal, soft_hard_different = 0, 0, 0
for path in sorted(FORMAL.glob('*_RAW.json.gz')):
    contract, fit = map(int, re.fullmatch(r'contract(\d+)_fit(\d+)_RAW.json.gz', path.name).groups())
    raw = json.loads(gzip.decompress(path.read_bytes()))
    assert len(raw['episodes']) == 128
    for index, row in enumerate(raw['episodes']):
        episode, results = row['episode'], row['results']
        assert episode['seed'] == 587103100+contract*100000+fit*1000+index
        assert episode['id'] == f'contract{contract}_fit{fit}_session{index:03d}'
        assert episode['id'] not in identifiers and episode['seed'] not in seeds
        identifiers.add(episode['id'])
        seeds.add(episode['seed'])
        assert len(results) == 5
        for filename, value in episode['files'].items():
            key = f'formal_confirmation01/episodes/{episode["id"]}/{filename}'
            assert expected[key] == value, key
            generator_files += 1
        for result in results.values():
            assert not result['failure']
            for event in result['events']:
                if 'decoded' in event:
                    assert event['decoded']['payload_sha256'] == episode['files'][event['packet']]
                    decoder_receipts += 1
        ordinary_equal += semantic(results['qvi_ordinary']) == semantic(results['qvi_analytic'])
        hard_qvi_equal += semantic(results['direct_hard_analytic']) == semantic(results['qvi_analytic'])
        soft_hard_different += semantic(results['direct_hard_analytic']) != semantic(results['direct_soft_analytic'])
assert len(identifiers) == 5120 and ordinary_equal == 5120
