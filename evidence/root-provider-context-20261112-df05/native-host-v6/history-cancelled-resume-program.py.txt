def tree_map(path,ref):
    raw=subprocess.check_output(['git','-C',str(path),'ls-tree','-r','-z',ref],env=env)
    result={}
    for entry in raw.split(b'\0'):
        if entry:
            meta,name=entry.split(b'\t',1);mode,kind,oid=meta.decode().split()
            result[name.decode()]=(mode,kind,oid)
    return result

def alive(pgid):
    try:os.killpg(pgid,0);return True
    except ProcessLookupError:return False
def command(path,args,log,source):
    started=time.time()
    with log.open('wb') as output:
        output.write((json.dumps({'source_commit':source,'cwd':str(path),'command':args,'started_unix':started})+'\n').encode());output.flush()
        process=subprocess.Popen(args,cwd=path,env=env,stdout=output,stderr=subprocess.STDOUT,start_new_session=True)
        rc=process.wait();residual=alive(process.pid)
        if residual:
            os.killpg(process.pid,signal.SIGTERM)
            deadline=time.monotonic()+10
            while alive(process.pid) and time.monotonic()<deadline:time.sleep(.1)
            if alive(process.pid):os.killpg(process.pid,signal.SIGKILL)
            deadline=time.monotonic()+5
            while alive(process.pid) and time.monotonic()<deadline:time.sleep(.1)
        result={'source_commit':source,'command':args,'exit_code':rc,'duration_seconds':time.time()-started,'process_group':process.pid,'residual_group_after_return':residual,'process_group_quiescent':not alive(process.pid)}
        output.write(('\n'+json.dumps(result)+'\n').encode());output.flush();os.fsync(output.fileno())
    emit({'phase':'command_finished',**result,'log':str(log)})
    return result

def oid_bytes(value):
    return hashlib.sha1(b'blob '+str(len(value)).encode()+b'\0'+value).hexdigest()
def ancestor_guard(root,path):
    if root.is_symlink() or not root.is_dir():
        raise RuntimeError('source root is not a real directory')
    relative=path.relative_to(root)
    parent=root
    for part in relative.parts[:-1]:
        parent=parent/part
        if parent.is_symlink():
            raise RuntimeError('source parent is a symlink: '+str(parent))
        if parent.exists() and not parent.is_dir():
            raise RuntimeError('source parent is not a directory: '+str(parent))
def physical(path,ref_or_entries):
    entries=tree_map(path,ref_or_entries) if isinstance(ref_or_entries,str) else ref_or_entries
    flags=subprocess.check_output(['git','-C',str(path),'ls-files','-v','-z'],env=env)
    hidden=[row[:1].decode()+' '+row[2:].decode() for row in flags.split(b'\0') if row and (row[:1]==b'S' or row[:1].islower())]
    if hidden:
        raise RuntimeError('index hides physical source paths: '+repr(hidden[:8]))
    for relative,(mode,kind,expected) in entries.items():
        if kind!='blob' or mode not in ('100644','100755','120000'):
            raise RuntimeError('unsupported source tree entry: '+relative+' '+mode+' '+kind)
        full=path/relative;ancestor_guard(path,full)
        info=full.lstat()
        if mode=='120000':
            if not stat.S_ISLNK(info.st_mode):
                raise RuntimeError('source symlink type mismatch: '+relative)
            actual=oid_bytes(os.fsencode(os.readlink(full)))
        else:
            if not stat.S_ISREG(info.st_mode):
                raise RuntimeError('source regular-file type mismatch: '+relative)
            actual_mode='100755' if info.st_mode & 0o111 else '100644'
            if actual_mode!=mode:
                raise RuntimeError('source executable mode mismatch: '+relative)
            digest=hashlib.sha1(b'blob '+str(info.st_size).encode()+b'\0')
            with full.open('rb') as file:
                while chunk:=file.read(1024*1024):
                    digest.update(chunk)
            after=full.lstat()
            signature=lambda item:(item.st_dev,item.st_ino,item.st_size,item.st_mtime_ns,item.st_ctime_ns,item.st_mode)
            if signature(info)!=signature(after):
                raise RuntimeError('source changed while being read: '+relative)
            actual=digest.hexdigest()
        if actual!=expected:
            raise RuntimeError('physical source blob mismatch: '+relative)
    return True
def materialize(path,expected,receipt_path):
    before_flags=subprocess.check_output(['git','-C',str(path),'ls-files','-v','-z'],env=env)
    tracked=subprocess.check_output(['git','-C',str(path),'ls-files','-z'],env=env)
    if not tracked:
        raise RuntimeError('empty tracked source inventory')
    before_index=git(path,'write-tree')
    config_args=['git','-C',str(path),'-c','core.sparseCheckout=false','-c','core.sparseCheckoutCone=false','-c','core.ignorestat=false']
    for option in ['--no-skip-worktree','--no-assume-unchanged']:
        subprocess.run(config_args+['update-index',option,'-z','--stdin'],input=tracked,env=env,check=True)
    missing=[]
    for raw in tracked.split(b'\0'):
        if not raw:
            continue
        relative=os.fsdecode(raw);full=path/relative;ancestor_guard(path,full)
        if not os.path.lexists(full):
            missing.append(raw)
    if missing:
        subprocess.run(config_args+['checkout-index','-z','--stdin'],input=b'\0'.join(missing)+b'\0',env=env,check=True)
    after_index=git(path,'write-tree')
    if after_index!=before_index:
        raise RuntimeError('materialization changed index source tree')
    physical(path,expected)
    after_flags=subprocess.check_output(['git','-C',str(path),'ls-files','-v','-z'],env=env)
    receipt={'source_directory':str(path),'index_tree_before':before_index,'index_tree_after':after_index,'tracked_entries':len(tracked.split(b'\0'))-1,'materialized_missing_paths':len(missing),'existing_files_overwritten':False,'shared_config_modified':False,'flags_before_sha256':hashlib.sha256(before_flags).hexdigest(),'flags_after_sha256':hashlib.sha256(after_flags).hexdigest(),'complete_physical_source_verified':True,'complete_expected_map_sha256':tree_digest(expected)}
    write(receipt_path,json.dumps(receipt,indent=2))
    emit({'phase':'materialized',**receipt})
