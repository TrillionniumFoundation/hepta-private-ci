def write(path,value):
    path.parent.mkdir(parents=True,exist_ok=True)
    blob=value.encode() if isinstance(value,str) else value
    with path.open('xb') as file:
        file.write(blob);file.flush();os.fsync(file.fileno())
    directory=path.parent
    while True:
        directory_fd=os.open(directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
        try:os.fsync(directory_fd)
        finally:os.close(directory_fd)
        if directory==directory.parent:break
        directory=directory.parent

def tree_map(path,ref):
    raw=subprocess.check_output(gitcmd(path,'ls-tree','-r','-z',ref),env=env)
    result={}
    for entry in raw.split(b'\0'):
        if entry:
            meta,name=entry.split(b'\t',1);mode,kind,oid=meta.decode().split()
            result[name.decode()]=(mode,kind,oid)
    return result
def sized_tree_map(path,ref):
    raw=subprocess.check_output(gitcmd(path,'ls-tree','-r','-l','-z','--full-tree',ref),env=env)
    result={}
    for entry in raw.split(b'\0'):
        if entry:
            meta,name=entry.split(b'\t',1);mode,kind,oid,size=meta.decode().split()
            if kind!='blob' or not size.isdigit():raise RuntimeError('unsupported source entry')
            result[name.decode()]=(mode,kind,oid,int(size))
    return result

def repository_guard(path):
    if path.is_symlink() or not path.is_dir():raise RuntimeError('source root is not a real directory')
    if pathlib.Path(plain_git(path,'rev-parse','--show-toplevel')).resolve()!=path.resolve():
        raise RuntimeError('Git worktree differs from explicit source root')
    if plain_git(path,'for-each-ref','--format=%(refname)','refs/replace/'):
        raise RuntimeError('replacement refs are not source identity')
    grafts=pathlib.Path(plain_git(path,'rev-parse','--git-path','info/grafts'))
    if not grafts.is_absolute():grafts=path/grafts
    if os.path.lexists(grafts):raise RuntimeError('grafted source history refused')
def flags_clear(path):
    flags=plain_git_bytes(path,'ls-files','-v','-z')
    if any(row and (row[:1]==b'S' or row[:1].islower()) for row in flags.split(b'\0')):
        raise RuntimeError('index hides tracked source')
    return flags
def ancestor_guard(root,path):
    if root.is_symlink() or not root.is_dir():raise RuntimeError('source root type')
    parent=root
    for part in path.relative_to(root).parts[:-1]:
        parent=parent/part
        if parent.is_symlink() or (parent.exists() and not parent.is_dir()):
            raise RuntimeError('source ancestor type: '+str(parent))

def _fingerprint(info: os.stat_result) -> tuple:
    return (info.st_dev, info.st_ino, info.st_mode, info.st_size,
            info.st_mtime_ns, info.st_ctime_ns)


def _tracked_blob(root_fd: int, path: bytes, mode: bytes, size: int) -> str:
    """Hash one raw blob through no-follow directory/file descriptors on CI Unix."""
    parts = path.split(b'/')
    if not parts or any(part in (b'', b'.', b'..') for part in parts):
        raise ValueError('noncanonical tracked path')
    parent = os.dup(root_fd)
    try:
        for part in parts[:-1]:
            next_parent = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW,
                                  dir_fd=parent)
            os.close(parent)
            parent = next_parent
        name = parts[-1]
        before = os.stat(name, dir_fd=parent, follow_symlinks=False)
        value = hashlib.sha1(b'blob ' + str(size).encode('ascii') + b'\0',
                             usedforsecurity=False)
        if mode == b'120000':
            if not stat.S_ISLNK(before.st_mode):
                raise ValueError('tracked symlink type changed')
            target = os.readlink(name, dir_fd=parent)
            if len(target) != size:
                raise ValueError('tracked symlink size changed')
            value.update(target)
        elif mode in (b'100644', b'100755'):
            if not stat.S_ISREG(before.st_mode) or before.st_size != size:
                raise ValueError('tracked regular file type or size changed')
            if bool(before.st_mode & 0o111) != (mode == b'100755'):
                raise ValueError('tracked executable mode changed')
            flags = os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK
            with os.fdopen(os.open(name, flags, dir_fd=parent), 'rb') as stream:
                if _fingerprint(os.fstat(stream.fileno())) != _fingerprint(before):
                    raise ValueError('tracked file changed before hashing')
                remaining = size
                while remaining:
                    chunk = stream.read(min(1024 * 1024, remaining))
                    if not chunk:
                        raise ValueError('tracked file truncated during hashing')
                    value.update(chunk)
                    remaining -= len(chunk)
                if stream.read(1) or _fingerprint(os.fstat(stream.fileno())) != _fingerprint(before):
                    raise ValueError('tracked file changed during hashing')
        else:
            # Gitlinks require a separate recursive-source contract; never skip them.
            raise ValueError('unsupported tracked entry mode')
        if _fingerprint(os.stat(name, dir_fd=parent, follow_symlinks=False)) != _fingerprint(before):
            raise ValueError('tracked path changed during hashing')
        check_parent=os.dup(root_fd)
        try:
            for part in parts[:-1]:
                next_parent=os.open(part,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=check_parent)
                os.close(check_parent);check_parent=next_parent
            if _fingerprint(os.stat(name,dir_fd=check_parent,follow_symlinks=False))!=_fingerprint(before):
                raise ValueError('tracked source path rebound during hashing')
        finally:
            os.close(check_parent)
        return value.hexdigest()
    finally:
        os.close(parent)



def physical(path,ref_or_entries):
    repository_guard(path);flags_clear(path)
    entries=sized_tree_map(path,ref_or_entries) if isinstance(ref_or_entries,str) else ref_or_entries
    root_before=path.lstat()
    root_fd=os.open(path,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    try:
        if _fingerprint(os.fstat(root_fd))!=_fingerprint(root_before):
            raise RuntimeError('source root changed before verification')
        total=0
        for relative,(mode,kind,expected,size) in entries.items():
            if kind!='blob':raise RuntimeError('unsupported source object')
            if _tracked_blob(root_fd,os.fsencode(relative),mode.encode(),size)!=expected:
                raise RuntimeError('physical source blob mismatch: '+relative)
            total+=size
        if _fingerprint(os.fstat(root_fd))!=_fingerprint(root_before) or _fingerprint(path.lstat())!=_fingerprint(root_before):
            raise RuntimeError('source root changed during verification')
    finally:
        os.close(root_fd)
    flags_clear(path);repository_guard(path)
    return {'tracked_entries':len(entries),'tracked_bytes':total,'complete_map_sha256':tree_digest(entries),'tracked_worktree_verified':True}

def _wait_without_reaping(child, deadline):
    """Observe exit under the original deadline without releasing the group ID."""
    while True:
        remaining=deadline-time.monotonic()
        if remaining<=0:raise ValueError('NATIVE_EXECUTOR_TIMEOUT')
        try:
            status=os.waitid(os.P_PID,child.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
        except InterruptedError:
            continue
        if status is not None:
            if status.si_code==os.CLD_EXITED:return status.si_status
            if status.si_code in (os.CLD_KILLED,os.CLD_DUMPED):return -status.si_status
            raise ValueError('NATIVE_EXECUTOR_WAIT_STATUS')
        time.sleep(min(remaining,.005))


def _darwin_only_exited_leader(child):
    """Disambiguate Darwin's zombie-only EPERM without waiving live denial.

    XNU killpg1 excludes zombies and can return EPERM for our unreaped leader.
    Its proc_listpids(PROC_PGRP_ONLY=2) walks both live and zombie lists under
    proc_list_lock. An eight-byte buffer returning exactly our single PID is
    therefore a complete singleton census, not a truncated list or size hint.
    Require WNOWAIT exit evidence first; zero/error/extra/foreign entries refuse.
    This trusts the system libproc/kernel and exclusive reaping, as the owner
    already trusts POSIX process isolation. It neither signals discovered PIDs
    nor manages descendants which left this session.
    """
    if sys.platform!='darwin' or child.returncode is not None:return False
    try:
        status=os.waitid(os.P_PID,child.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
        if (status is None or status.si_pid!=child.pid or
                status.si_code not in (os.CLD_EXITED,os.CLD_KILLED,os.CLD_DUMPED)):
            return False
        import ctypes
        library=ctypes.CDLL('/usr/lib/libproc.dylib',use_errno=True)
        census=library.proc_listpids
        census.argtypes=[ctypes.c_uint32,ctypes.c_uint32,ctypes.c_void_p,ctypes.c_int]
        census.restype=ctypes.c_int
        pids=(ctypes.c_int*2)()
        ctypes.set_errno(0)
        size=census(2,child.pid,pids,ctypes.sizeof(pids))
        return (size==ctypes.sizeof(ctypes.c_int) and pids[0]==child.pid
                and ctypes.get_errno()==0)
    except (OSError,AttributeError,ValueError):
        return False


def owned_run(command: list[str], log: Path, *, timeout: int = 900,
        env: dict[str, str] | None = None, cwd: Path = ROOT,
        expected_source: str | None = None) -> dict:
    if not all(hasattr(os, name) for name in (
            'waitid', 'P_PID', 'WEXITED', 'WNOHANG', 'WNOWAIT',
            'CLD_EXITED', 'CLD_KILLED', 'CLD_DUMPED', 'killpg')):
        raise ValueError('CI_PROCESS_PLATFORM')
    started = time.monotonic_ns()
    timed_out = False
    executed = False
    process_code = None
    source_error = None
    source_phase = None
    code = 125
    with log.open('xb') as stream:
        if expected_source is not None:
            source_error = _source_error(expected_source)
            if source_error is not None:
                source_phase = 'before'
        if source_error is None:
            process = subprocess.Popen(command, cwd=cwd, env=env, stdout=stream,
                                       stderr=subprocess.STDOUT, start_new_session=True)
            executed = True
            try:
                try:
                    code = _wait_without_reaping(process, time.monotonic() + timeout)
                except ValueError as error:
                    if str(error) != 'NATIVE_EXECUTOR_TIMEOUT':
                        raise
                    timed_out = True
                    try:
                        os.killpg(process.pid, signal.SIGTERM)
                    except ProcessLookupError:
                        pass
                    except PermissionError:
                        if not _darwin_only_exited_leader(process):
                            raise
                    try:
                        _wait_without_reaping(process, time.monotonic() + 10)
                    except ValueError as error:
                        if str(error) != 'NATIVE_EXECUTOR_TIMEOUT':
                            raise
                    code = 124
            finally:
                # Keep the leader unreaped through the LAST group signal. A
                # successful or nonzero leader exit can still leave log writers.
                # Only this created group is signalled, never discovered PIDs.
                # The caller must remain the sole reaper; escaped sessions and
                # kernel cleanup latency remain outside this owner contract.
                try:
                    try:
                        os.killpg(process.pid, signal.SIGKILL)
                    except ProcessLookupError:
                        pass
                    except PermissionError:
                        if not _darwin_only_exited_leader(process):
                            raise
                finally:
                    process.wait()
            process_code = process.returncode
    if executed and expected_source is not None:
        source_error = _source_error(expected_source)
        if source_error is not None:
            source_phase = 'after'
            # Keep the original nonzero process/timeout outcome. Only an
            # otherwise successful command needs a distinct source refusal.
            if code == 0:
                code = 125
    result = {'command': command, 'cwd': str(cwd), 'exit_code': code,
              'timed_out': timed_out, 'elapsed_ns': time.monotonic_ns() - started,
              'log': log.name, 'executed': executed,
              'process_exit_code': process_code}
    if expected_source is not None:
        result.update(expected_source=expected_source,
                      source_before_verified=source_phase != 'before',
                      source_after_verified=executed and source_error is None,
                      source_verified=executed and source_error is None)
        if source_error is not None:
            result.update(source_error=source_error, source_failure_phase=source_phase)
    return result

def materialize(path,expected,receipt_path):

    for option in ['--no-skip-worktree','--no-assume-unchanged']:
        subprocess.run(gitcmd(path,'update-index',option,'-z','--stdin'),input=tracked,env=env,check=True)
    missing=[]
    for raw in tracked.split(b'\0'):
        if raw:
            full=path/os.fsdecode(raw);ancestor_guard(path,full)
            if not os.path.lexists(full):missing.append(raw)

    if missing:
        subprocess.run(gitcmd(path,'checkout-index','-z','--stdin'),input=b'\0'.join(missing)+b'\0',env=env,check=True)
    for option in ['--no-skip-worktree','--no-assume-unchanged']:
        subprocess.run(gitcmd(path,'update-index',option,'-z','--stdin'),input=tracked,env=env,check=True)
    if git(path,'write-tree')!=before_index:raise RuntimeError('materialization changed index source tree')

             'tracked_entries':len(tracked.split(b'\0'))-1,'materialized_missing_paths':len(missing),
             'existing_files_overwritten':False,'shared_config_modified':False,
             'flags_before_sha256':hashlib.sha256(before_flags).hexdigest(),'flags_after_sha256':hashlib.sha256(after_flags).hexdigest(),
             'physical_source':proof}
    write(receipt_path,json.dumps(receipt,indent=2));emit({'phase':'materialized',**receipt})
