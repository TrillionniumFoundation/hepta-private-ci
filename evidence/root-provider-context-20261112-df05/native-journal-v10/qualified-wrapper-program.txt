def parent_list(path, commit):
    return git(path, 'show', '-s', '--format=%P', commit).split()

def apply_candidate(path, parent, selected, message):
    clean(path, parent)
    expected = sized_tree_map(path, parent)
    for relative in selected:
        if relative not in files or relative not in expected:
            raise RuntimeError('candidate overlay is not an existing reviewed source entry')
        p = path / relative
        ancestor_guard(path, p)
        before = p.lstat()
        if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1:
            raise RuntimeError('candidate source entry type changed')
        payload = files[relative].encode('utf-8')
        fd = os.open(p, os.O_WRONLY | os.O_TRUNC | os.O_NOFOLLOW)
        with os.fdopen(fd, 'wb') as stream:
            stream.write(payload)
            stream.flush()
            os.fsync(stream.fileno())
        expected[relative] = (expected[relative][0], 'blob', oid_bytes(payload), len(payload))
    physical(path, expected)
    visible = set(plain_git(path, 'diff', '--name-only').splitlines()) | set(plain_git(path, 'ls-files', '--others', '--exclude-standard').splitlines())
    if visible != set(selected):
        raise RuntimeError('candidate overlay changed unexpected source')
    git(path, 'add', '--', *selected)
    if set(git(path, 'diff', '--cached', '--name-only').splitlines()) != set(selected):
        raise RuntimeError('source commit staged allowlist mismatch')
    frozen = git(path, 'write-tree')
    if sized_tree_map(path, frozen) != expected:
        raise RuntimeError('candidate index differs from complete expected source')
    git(path, 'commit', '-m', message)
    source = git(path, 'rev-parse', 'HEAD')
    if parent_list(path, source) != [parent] or sized_tree_map(path, source) != expected:
        raise RuntimeError('source commit parent or complete tree differs')
    clean(path, source)
    return source

def merge_journal(journal_evidence):
    path = host_dir
    left = old_evidence['host']
    clean(path, left)
    if git(path, 'merge-base', left, journal_evidence) != old_sources['journal']:
        raise RuntimeError('unexpected merge base')
    original = tree_map(path, old_sources['journal'])
    left_map = tree_map(path, left)
    right_map = tree_map(path, journal_evidence)
    expected = {}
    for relative in sorted(set(original) | set(left_map) | set(right_map)):
        common, a, b = original.get(relative), left_map.get(relative), right_map.get(relative)
        if a == b:
            value = a
        elif a == common:
            value = b
        elif b == common:
            value = a
        else:
            raise RuntimeError('reviewed file-level merge is not conflict free: ' + relative)
        if value is not None:
            expected[relative] = value
    no_hold()
    result = subprocess.run(gitcmd(path, 'merge', '--no-ff', '--no-commit', journal_evidence), env=env, capture_output=True)
    write(evidence / 'host-merge.log', result.stdout + result.stderr)
    if result.returncode:
        raise RuntimeError('merge conflict or error; no automatic resolution')
    if git(path, 'rev-parse', 'MERGE_HEAD') != journal_evidence:
        raise RuntimeError('merge target changed')
    if tree_map(path, git(path, 'write-tree')) != expected or plain_git(path, 'diff', '--name-only'):
        raise RuntimeError('merge index differs from complete expected union')
    physical(path, git(path, 'write-tree'))
    git(path, 'commit', '-m', 'merge: include validated provider context journal and evidence')
    merged = git(path, 'rev-parse', 'HEAD')
    if parent_list(path, merged) != [left, journal_evidence] or tree_map(path, merged) != expected:
        raise RuntimeError('actual merge parent order or complete tree changed')
    clean(path, merged)
    proof = {
        'merge_commit': merged, 'parents': [left, journal_evidence],
        'merge_base': old_sources['journal'], 'complete_expected_map_sha256': tree_digest(expected),
        'complete_mode_blob_union_verified': True, 'physical_source': verify_source(path, merged),
    }
    write(evidence / 'host-merge-source.json', json.dumps(proof, indent=2))
    emit({'phase': 'actual_merge_frozen', **proof})
    return merged

def source_line_count(path, from_commit, to_commit):
    raw = git(path, 'diff', '--numstat', from_commit, to_commit, '--', 'codex-rs')
    count = 0
    for line in raw.splitlines():
        added, deleted, name = line.split('\t', 2)
        if not added.isdigit() or not deleted.isdigit():
            raise RuntimeError('binary source diff is outside scope')
        count += int(added) + int(deleted)
    if count > 800:
        raise RuntimeError('nonmechanical source stage exceeds the reviewed size limit')
    return count

def snapshot_hooks(path):
    if not os.path.lexists(path):
        return {'path': str(path), 'exists': False, 'entries': {}}
    if path.is_symlink() or not path.is_dir():
        raise RuntimeError('original hook directory has unsupported type')
    entries = {}
    for item in sorted(path.iterdir()):
        info = item.lstat()
        value = {'mode': oct(stat.S_IMODE(info.st_mode)), 'uid': info.st_uid, 'gid': info.st_gid}
        if stat.S_ISREG(info.st_mode):
            value.update(kind='file', bytes=info.st_size, sha256=hashlib.sha256(item.read_bytes()).hexdigest())
        elif stat.S_ISLNK(info.st_mode):
            value.update(kind='symlink', target=os.readlink(item))
        elif stat.S_ISDIR(info.st_mode):
            value.update(kind='directory')
        else:
            raise RuntimeError('original hooks contain unsupported object')
        entries[item.name] = value
    return {'path': str(path), 'exists': True, 'entries': entries}

def identity_map(raw):
    result = {}
    for entry in raw.split(b'\0'):
        if entry:
            key, value = entry.split(b'\n', 1)
            result[key.decode('utf-8')] = value.decode('utf-8')
    return result

def effective_identity(path):
    result = subprocess.run(gitcmd(path, 'config', '--null', '--get-regexp', identity_pattern), env=env, capture_output=True)
    if result.returncode not in (0, 1):
        raise RuntimeError('effective native identity could not be read')
    return identity_map(result.stdout)

def read_prior(spec, exact_path):
    path = pathlib.Path(spec['path'])
    if path != exact_path:
        raise RuntimeError('private original path changed')
    before = path.lstat()
    if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1:
        raise RuntimeError('private original type or link count changed')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        opened = os.fstat(fd)
        with os.fdopen(os.dup(fd), 'rb') as stream:
            payload = stream.read()
        if _fingerprint(before) != _fingerprint(opened) or _fingerprint(opened) != _fingerprint(os.fstat(fd)) or _fingerprint(opened) != _fingerprint(path.lstat()):
            raise RuntimeError('private original changed while being read')
    finally:
        os.close(fd)
    if len(payload) != spec['bytes'] or hashlib.sha256(payload).hexdigest() != spec['sha256']:
        raise RuntimeError('private original byte identity changed')
    return payload
